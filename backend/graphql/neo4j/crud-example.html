CRUD Operations with Neo4j GraphQL

1. Project Setup

Initialize Node.js Project

# Create project directory
mkdir neo4j-graphql-crud
cd neo4j-graphql-crud

# Initialize npm project
npm init -y

# Install dependencies
npm install @neo4j/graphql neo4j-driver apollo-server graphql dotenv

Project Structure

neo4j-graphql-crud/
├── .env
├── package.json
├── index.js
└── schema.js
2. Environment Configuration

Create .env file with your Neo4j credentials:

# .env
NEO4J_URI=bolt://localhost:7687
NEO4J_USER=neo4j
NEO4J_PASSWORD=your_password_here

# For Neo4j AuraDB, use:
# NEO4J_URI=neo4j+s://xxxxx.databases.neo4j.io
# NEO4J_USER=neo4j
# NEO4J_PASSWORD=your_aura_password
3. Define GraphQL Schema

Create schema.js with type definitions:

// schema.js
const { gql } = require('apollo-server');

const typeDefs = gql`
  type User {
    id: ID! @id
    name: String!
    email: String! @unique
    age: Int
    createdAt: DateTime! @timestamp(operations: [CREATE])
    updatedAt: DateTime! @timestamp(operations: [UPDATE])
    posts: [Post!]! @relationship(type: "POSTED", direction: OUT)
  }

  type Post {
    id: ID! @id
    title: String!
    content: String!
    published: Boolean! @default(value: false)
    createdAt: DateTime! @timestamp(operations: [CREATE])
    updatedAt: DateTime! @timestamp(operations: [UPDATE])
    author: User! @relationship(type: "POSTED", direction: IN)
    comments: [Comment!]! @relationship(type: "HAS_COMMENT", direction: OUT)
  }

  type Comment {
    id: ID! @id
    text: String!
    createdAt: DateTime! @timestamp(operations: [CREATE])
    post: Post! @relationship(type: "HAS_COMMENT", direction: IN)
    author: User! @relationship(type: "WROTE", direction: IN)
  }
`;

module.exports = typeDefs;
4. Create GraphQL Server

Create index.js with server setup:

// index.js
require('dotenv').config();
const { ApolloServer } = require('apollo-server');
const neo4j = require('neo4j-driver');
const { Neo4jGraphQL } = require('@neo4j/graphql');
const typeDefs = require('./schema');

// Create Neo4j driver instance
const driver = neo4j.driver(
  process.env.NEO4J_URI,
  neo4j.auth.basic(process.env.NEO4J_USER, process.env.NEO4J_PASSWORD)
);

// Create Neo4jGraphQL instance
const neoSchema = new Neo4jGraphQL({ 
  typeDefs, 
  driver,
  // Enable debug to see generated Cypher queries
  debug: true
});

// Start server
async function startServer() {
  try {
    // Verify database connectivity
    await driver.verifyConnectivity();
    console.log('✓ Connected to Neo4j');

    // Generate executable schema
    const schema = await neoSchema.getSchema();

    // Create Apollo Server
    const server = new ApolloServer({
      schema,
      context: ({ req }) => ({ req }),
    });

    // Start listening
    const { url } = await server.listen(4000);
    console.log(`🚀 Server ready at ${url}`);
  } catch (error) {
    console.error('Failed to start server:', error);
    process.exit(1);
  }
}

// Handle graceful shutdown
process.on('SIGINT', async () => {
  console.log('\nShutting down...');
  await driver.close();
  process.exit(0);
});

startServer();
5. Run the Server
# Start the server
node index.js

# Output:
# ✓ Connected to Neo4j
# 🚀 Server ready at http://localhost:4000/

Open http://localhost:4000 in your browser to access Apollo Studio.

6. CREATE Operations

Create a User

mutation CreateUser {
  createUsers(
    input: [
      {
        name: "Alice Johnson"
        email: "alice@example.com"
        age: 30
      }
    ]
  ) {
    users {
      id
      name
      email
      age
      createdAt
    }
  }
}

Create User with Posts (Nested Creation)

mutation CreateUserWithPosts {
  createUsers(
    input: [
      {
        name: "Bob Smith"
        email: "bob@example.com"
        age: 25
        posts: {
          create: [
            {
              node: {
                title: "Getting Started with Neo4j"
                content: "Neo4j is a powerful graph database..."
                published: true
              }
            }
            {
              node: {
                title: "GraphQL Best Practices"
                content: "Here are some tips for building GraphQL APIs..."
                published: false
              }
            }
          ]
        }
      }
    ]
  ) {
    users {
      id
      name
      posts {
        id
        title
        published
      }
    }
  }
}

Create Comment

mutation CreateComment {
  createComments(
    input: [
      {
        text: "Great article! Very helpful."
        post: {
          connect: {
            where: { node: { id: "post-id-here" } }
          }
        }
        author: {
          connect: {
            where: { node: { email: "alice@example.com" } }
          }
        }
      }
    ]
  ) {
    comments {
      id
      text
      createdAt
      author {
        name
      }
      post {
        title
      }
    }
  }
}
7. READ Operations

Get All Users

query GetAllUsers {
  users {
    id
    name
    email
    age
    createdAt
  }
}

Get User by Email with Posts

query GetUserByEmail {
  users(where: { email: "bob@example.com" }) {
    id
    name
    email
    posts {
      id
      title
      content
      published
      createdAt
      comments {
        text
        author {
          name
        }
      }
    }
  }
}

Get Posts with Filtering and Sorting

query GetPublishedPosts {
  posts(
    where: { published: true }
    options: { 
      sort: [{ createdAt: DESC }]
      limit: 10
    }
  ) {
    id
    title
    content
    createdAt
    author {
      name
      email
    }
    comments {
      text
      author {
        name
      }
    }
  }
}

Search Posts by Title

query SearchPosts {
  posts(where: { title_CONTAINS: "Neo4j" }) {
    id
    title
    author {
      name
    }
  }
}

Get User Count

query GetUserCount {
  usersAggregate {
    count
  }
}
8. UPDATE Operations

Update User Information

mutation UpdateUser {
  updateUsers(
    where: { email: "alice@example.com" }
    update: { 
      name: "Alice Johnson-Smith"
      age: 31
    }
  ) {
    users {
      id
      name
      age
      updatedAt
    }
  }
}

Update Post and Publish

mutation PublishPost {
  updatePosts(
    where: { id: "post-id-here" }
    update: { 
      published: true
      content: "Updated content here..."
    }
  ) {
    posts {
      id
      title
      published
      updatedAt
    }
  }
}

Connect Existing Nodes (Add Relationship)

mutation AddPostToUser {
  updateUsers(
    where: { email: "alice@example.com" }
    connect: {
      posts: {
        where: { node: { id: "existing-post-id" } }
      }
    }
  ) {
    users {
      name
      posts {
        title
      }
    }
  }
}

Disconnect Nodes (Remove Relationship)

mutation RemovePostFromUser {
  updateUsers(
    where: { email: "alice@example.com" }
    disconnect: {
      posts: {
        where: { node: { id: "post-id-to-remove" } }
      }
    }
  ) {
    users {
      name
      posts {
        title
      }
    }
  }
}

10. Advanced Queries

Pagination with Cursor

query GetUsersWithPagination {
  usersConnection(
    first: 10
    after: "cursor-string-here"
    sort: [{ name: ASC }]
  ) {
    edges {
      cursor
      node {
        id
        name
        email
      }
    }
    pageInfo {
      hasNextPage
      hasPreviousPage
      startCursor
      endCursor
    }
    totalCount
  }
}

Complex Filtering

query ComplexFilter {
  users(
    where: {
      OR: [
        { age_GT: 25 }
        { name_CONTAINS: "Smith" }
      ]
      AND: [
        { email_ENDS_WITH: "@example.com" }
      ]
    }
  ) {
    name
    email
    age
  }
}

Aggregations

query GetUserStats {
  users {
    name
    postsAggregate {
      count
      node {
        title {
          longest
          shortest
        }
      }
    }
  }
}

12. Viewing Data in Neo4j Browser

Open Neo4j Browser at http://localhost:7474 and run Cypher queries to see your data:

// View all users and their posts
MATCH (u:User)-[:POSTED]->(p:Post)
RETURN u, p

// View complete graph
MATCH (n)
RETURN n
LIMIT 100

// Count nodes by type
MATCH (u:User) RETURN count(u) as users
MATCH (p:Post) RETURN count(p) as posts
MATCH (c:Comment) RETURN count(c) as comments
