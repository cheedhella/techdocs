3. Define Schema with Subscriptions

Create server/schema.js:

// server/schema.js
const { gql } = require('apollo-server');

const typeDefs = gql`
  type Message {
    id: ID! @id
    text: String!
    username: String!
    createdAt: DateTime! @timestamp(operations: [CREATE])
    room: Room! @relationship(type: "IN_ROOM", direction: OUT)
  }

  type Room {
    id: ID! @id
    name: String! @unique
    createdAt: DateTime! @timestamp(operations: [CREATE])
    messages: [Message!]! @relationship(type: "IN_ROOM", direction: IN)
  }

  type User {
    id: ID! @id
    username: String! @unique
    online: Boolean! @default(value: false)
    lastSeen: DateTime
    createdAt: DateTime! @timestamp(operations: [CREATE])
  }

  type Notification {
    id: ID!
    type: NotificationType!
    message: String!
    timestamp: DateTime!
    user: User
  }

  enum NotificationType {
    MESSAGE
    USER_JOINED
    USER_LEFT
    ROOM_CREATED
  }

  # Custom mutations for real-time features
  type Mutation {
    sendMessage(roomId: ID!, username: String!, text: String!): Message!
    joinRoom(roomId: ID!, username: String!): Room!
    leaveRoom(roomId: ID!, username: String!): Boolean!
    setUserOnline(username: String!, online: Boolean!): User!
  }

  # Subscriptions for live updates
  type Subscription {
    messageAdded(roomId: ID!): Message!
    userJoined(roomId: ID!): User!
    userLeft(roomId: ID!): User!
    userStatusChanged(username: String!): User!
    roomCreated: Room!
    notification(username: String!): Notification!
  }
`;

module.exports = typeDefs;
4. Implement Custom Resolvers

Create server/resolvers.js:

// server/resolvers.js
const { PubSub } = require('graphql-subscriptions');

const pubsub = new PubSub();

// Subscription event names
const MESSAGE_ADDED = 'MESSAGE_ADDED';
const USER_JOINED = 'USER_JOINED';
const USER_LEFT = 'USER_LEFT';
const USER_STATUS_CHANGED = 'USER_STATUS_CHANGED';
const ROOM_CREATED = 'ROOM_CREATED';
const NOTIFICATION = 'NOTIFICATION';

const resolvers = {
  Mutation: {
    sendMessage: async (_, { roomId, username, text }, { driver }) => {
      const session = driver.session();
      try {
        // Create message in Neo4j
        const result = await session.run(
          `
          MATCH (r:Room {id: $roomId})
          CREATE (m:Message {
            id: randomUUID(),
            text: $text,
            username: $username,
            createdAt: datetime()
          })
          CREATE (m)-[:IN_ROOM]->(r)
          RETURN m, r
          `,
          { roomId, text, username }
        );

        const message = result.records[0].get('m').properties;
        const room = result.records[0].get('r').properties;

        // Convert Neo4j DateTime to ISO string
        message.createdAt = message.createdAt.toString();

        // Publish to subscribers
        pubsub.publish(MESSAGE_ADDED, {
          messageAdded: message,
          roomId,
        });

        // Send notification
        pubsub.publish(NOTIFICATION, {
          notification: {
            id: Math.random().toString(),
            type: 'MESSAGE',
            message: `${username} sent a message in ${room.name}`,
            timestamp: new Date().toISOString(),
            user: { username },
          },
          username: null, // Broadcast to all
        });

        return message;
      } finally {
        await session.close();
      }
    },

    joinRoom: async (_, { roomId, username }, { driver }) => {
      const session = driver.session();
      try {
        // Get or create user and room
        const result = await session.run(
          `
          MERGE (u:User {username: $username})
          ON CREATE SET u.id = randomUUID(), u.createdAt = datetime()
          SET u.online = true, u.lastSeen = datetime()
          
          MATCH (r:Room {id: $roomId})
          
          RETURN u, r
          `,
          { roomId, username }
        );

        const user = result.records[0].get('u').properties;
        const room = result.records[0].get('r').properties;

        // Publish user joined event
        pubsub.publish(USER_JOINED, {
          userJoined: user,
          roomId,
        });

        // Send notification
        pubsub.publish(NOTIFICATION, {
          notification: {
            id: Math.random().toString(),
            type: 'USER_JOINED',
            message: `${username} joined ${room.name}`,
            timestamp: new Date().toISOString(),
            user,
          },
          username: null,
        });

        return room;
      } finally {
        await session.close();
      }
    },

    leaveRoom: async (_, { roomId, username }, { driver }) => {
      const session = driver.session();
      try {
        await session.run(
          `
          MATCH (u:User {username: $username})
          SET u.online = false, u.lastSeen = datetime()
          `,
          { username }
        );

        const user = { username, online: false };

        // Publish user left event
        pubsub.publish(USER_LEFT, {
          userLeft: user,
          roomId,
        });

        return true;
      } finally {
        await session.close();
      }
    },

    setUserOnline: async (_, { username, online }, { driver }) => {
      const session = driver.session();
      try {
        const result = await session.run(
          `
          MATCH (u:User {username: $username})
          SET u.online = $online, u.lastSeen = datetime()
          RETURN u
          `,
          { username, online }
        );

        const user = result.records[0].get('u').properties;

        // Publish status change
        pubsub.publish(USER_STATUS_CHANGED, {
          userStatusChanged: user,
          username,
        });

        return user;
      } finally {
        await session.close();
      }
    },
  },

  Subscription: {
    messageAdded: {
      subscribe: (_, { roomId }) => {
        return pubsub.asyncIterator([MESSAGE_ADDED]);
      },
      resolve: (payload, { roomId }) => {
        // Filter messages by room
        if (payload.roomId === roomId) {
          return payload.messageAdded;
        }
        return null;
      },
    },

    userJoined: {
      subscribe: (_, { roomId }) => {
        return pubsub.asyncIterator([USER_JOINED]);
      },
      resolve: (payload, { roomId }) => {
        if (payload.roomId === roomId) {
          return payload.userJoined;
        }
        return null;
      },
    },

    userLeft: {
      subscribe: (_, { roomId }) => {
        return pubsub.asyncIterator([USER_LEFT]);
      },
      resolve: (payload, { roomId }) => {
        if (payload.roomId === roomId) {
          return payload.userLeft;
        }
        return null;
      },
    },

    userStatusChanged: {
      subscribe: (_, { username }) => {
        return pubsub.asyncIterator([USER_STATUS_CHANGED]);
      },
      resolve: (payload, { username }) => {
        if (!username || payload.username === username) {
          return payload.userStatusChanged;
        }
        return null;
      },
    },

    roomCreated: {
      subscribe: () => pubsub.asyncIterator([ROOM_CREATED]),
    },

    notification: {
      subscribe: () => pubsub.asyncIterator([NOTIFICATION]),
      resolve: (payload, { username }) => {
        // If username specified, filter; otherwise broadcast
        if (!username || !payload.username || payload.username === username) {
          return payload.notification;
        }
        return null;
      },
    },
  },
};

module.exports = { resolvers, pubsub };
5. Create Server with WebSocket Support

Create server/index.js:

// server/index.js
require('dotenv').config();
const { ApolloServer } = require('apollo-server');
const neo4j = require('neo4j-driver');
const { Neo4jGraphQL } = require('@neo4j/graphql');
const typeDefs = require('./schema');
const { resolvers } = require('./resolvers');

// Create Neo4j driver
const driver = neo4j.driver(
  process.env.NEO4J_URI,
  neo4j.auth.basic(process.env.NEO4J_USER, process.env.NEO4J_PASSWORD)
);

async function startServer() {
  try {
    // Verify connectivity
    await driver.verifyConnectivity();
    console.log('✓ Connected to Neo4j');

    // Create Neo4jGraphQL instance
    const neoSchema = new Neo4jGraphQL({
      typeDefs,
      driver,
      resolvers, // Add custom resolvers
    });

    // Generate schema
    const schema = await neoSchema.getSchema();

    // Create Apollo Server with subscription support
    const server = new ApolloServer({
      schema,
      context: ({ req }) => ({ req, driver }),
      subscriptions: {
        path: '/subscriptions',
        onConnect: (connectionParams) => {
          console.log('Client connected to subscriptions');
          return { driver };
        },
        onDisconnect: () => {
          console.log('Client disconnected from subscriptions');
        },
      },
    });

    // Start server
    const { url, subscriptionsUrl } = await server.listen(
      process.env.PORT || 4000
    );

    console.log(`🚀 Server ready at ${url}`);
    console.log(`🔔 Subscriptions ready at ${subscriptionsUrl}`);

    // Create sample room if not exists
    await createSampleData();
  } catch (error) {
    console.error('Failed to start server:', error);
    process.exit(1);
  }
}

async function createSampleData() {
  const session = driver.session();
  try {
    await session.run(
      `
      MERGE (r:Room {name: "General"})
      ON CREATE SET r.id = randomUUID(), r.createdAt = datetime()
      RETURN r
      `
    );
    console.log('✓ Sample room created');
  } finally {
    await session.close();
  }
}

// Graceful shutdown
process.on('SIGINT', async () => {
  console.log('\nShutting down...');
  await driver.close();
  process.exit(0);
});

startServer();

    const WS_URL = 'ws://localhost:4000/subscriptions';
    const HTTP_URL = 'http://localhost:4000/graphql';
    const ROOM_ID = 'general-room-id'; // Replace with actual room ID
    const USERNAME = 'User' + Math.floor(Math.random() * 1000);

    let ws;
    let subscriptionId = 1;

    // Initialize WebSocket connection
    function connect() {
      ws = new WebSocket(WS_URL, 'graphql-ws');

      ws.onopen = () => {
        console.log('WebSocket connected');
        ws.send(JSON.stringify({ type: 'connection_init' }));
      };

      ws.onmessage = (event) => {
        const message = JSON.parse(event.data);
        handleMessage(message);
      };

      ws.onerror = (error) => {
        console.error('WebSocket error:', error);
        updateStatus('disconnected');
      };

      ws.onclose = () => {
        console.log('WebSocket disconnected');
        updateStatus('disconnected');
        setTimeout(connect, 3000); // Reconnect
      };

    function handleMessage(message) {
      switch (message.type) {
        case 'connection_ack':
          console.log('Connection acknowledged');
          updateStatus('connected');
          subscribeToMessages();
          joinRoom();
          break;
        case 'data':
          if (message.payload.data.messageAdded) {
            displayMessage(message.payload.data.messageAdded);
          }
          break;
      }

    function subscribeToMessages() {

                id
                text
                username
                createdAt
              }

          `,

        }

      ws.send(JSON.stringify(subscription));
    }

            id
            text
            username
            createdAt
          }
        }

        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          query: mutation,
          variables: { roomId: ROOM_ID, username: USERNAME, text }
        })

12. Production Considerations

Use Redis PubSub for Scaling

# Install Redis PubSub
npm install graphql-redis-subscriptions ioredis
// Replace in-memory PubSub with Redis
const { RedisPubSub } = require('graphql-redis-subscriptions');
const Redis = require('ioredis');

const options = {
  host: process.env.REDIS_HOST || 'localhost',
  port: process.env.REDIS_PORT || 6379,
  retryStrategy: times => Math.min(times * 50, 2000),
};

const pubsub = new RedisPubSub({
  publisher: new Redis(options),
  subscriber: new Redis(options),
});

Authentication for Subscriptions

subscriptions: {
  onConnect: (connectionParams) => {
    if (connectionParams.authToken) {
      // Verify JWT token
      const user = verifyToken(connectionParams.authToken);
      return { user, driver };
    }
    throw new Error('Missing auth token');
  },
}

Rate Limiting

  • Limit subscription connections per user
  • Throttle message publishing frequency
  • Set maximum subscription duration

Monitoring

  • Track active WebSocket connections
  • Monitor subscription performance
  • Log subscription errors
  • Alert on connection spikes

14. Troubleshooting

WebSocket Connection Issues

  • Check firewall settings allow WebSocket connections
  • Verify correct WebSocket URL (ws:// or wss://)
  • Ensure server supports WebSocket upgrades

Subscriptions Not Receiving Updates

  • Verify pubsub.publish() is called after mutations
  • Check subscription filters match published data
  • Ensure WebSocket connection is active

Performance Issues

  • Use Redis PubSub instead of in-memory
  • Implement subscription filtering server-side
  • Limit subscription payload size
  • Add connection limits per user
