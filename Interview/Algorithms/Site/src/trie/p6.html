Given a Trie and word, can you find if the given word can be formed by combing any 2 words in the Trie? 
        
public class Exercise4 {
    public static void main(String[] args) {
        Trie trie = getTrie();
        Exercise4.isFormationPossible(trie, "thereby");
    }

    public static boolean isFormationPossible(Trie trie, String word) {
        for (int i = 0; i < word.length(); i++) {
            // Slice the word into two strings in each iteration
            String first = word.substring(0, i);
            String second = word.substring(i, word.length());

            // If both substrings are present in the trie, the condition is fulfilled
            if (trie.contains(first) && trie.contains(second)) {
                System.out.println(first);
                System.out.println(second);
                return true;
            }
        }
        return false;
    }

    private static Trie getTrie() {
        Trie d1 = new Trie();
        d1.insert("the");
        d1.insert("a");
        d1.insert("there");
        d1.insert("answer");
        d1.insert("any");
        d1.insert("by");
        d1.insert("bye");
        d1.insert("their");
        return d1;
    }
}

// Output:
there
by
