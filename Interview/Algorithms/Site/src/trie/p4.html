public List getAllWords() {
    ArrayList allWords = new ArrayList();
    getAllWords(root, allWords, "");
    return allWords;
}

private void getAllWords(Node current, List allWords, String currentWord) {
    if (current.isWord()) {
        allWords.add(currentWord);
    }
    for (Entry entry : current.getChildren().entrySet()) {
        String newWord = currentWord + Character.toString(entry.getKey());
        getAllWords(entry.getValue(), allWords, newWord);
    }
}
