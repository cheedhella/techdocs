public class BinarySearch {
    public static void main(String[] args) {
        int[] data = { 1, 3, 9, 10, 12, 15, 19, 22 };
        int target = 19;
        System.out.println(target + " found at index: " + iterativeSearch(data, target));
        int index = recursiveSearch(data, target, 0, data.length);
        System.out.println(target + " found at index: " + index);
    }

    // Time Complexity: O(log n); Space Complexity: O(1);
    private static int iterativeSearch(int[] data, int target) {
        int left = 0, right = data.length;
        while (left < right) {
            int mid = (left + right) / 2;
            if (data[mid] == target) {
                return mid;
            }
            if (target < data[mid]) {
                right = mid - 1; // target el is in right sub array!
            }
            else {
                left = mid + 1; // target el is in right sub array!
            }
        }
        return -1;
    }

    // Time Complexity: O(log n); Space Complexity: O(log n);
    private static int recursiveSearch(int[] data, int target, int left, int right) {
        if (left > right)
            return -1;
        int mid = (left + right) / 2;
        if (data[mid] == target) {
            return mid;
        }
        if (target < data[mid]) {
            return recursiveSearch(data, target, left, mid - 1); // target el is in right sub array!
        }
        return recursiveSearch(data, target, mid + 1, right);
    }
}

// Output:
19 found at index: 6
19 found at index: 6
