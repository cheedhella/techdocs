import java.util.ArrayList;
import java.util.Arrays;

// Heapify takes O(N) time;
// Removal of each min takes O(logN) time; K removals take O(K*logN) time;
// Total time complexity: O(N + K*logN);
public class MinKApp {
    public static void main(String[] args) {
        ArrayList data = new ArrayList();
        data.add(1);
        data.add(3);
        data.add(9);
        data.add(15);
        data.add(12);
        data.add(5);
        System.out.println("Input Array: " + Arrays.toString(data.toArray()));

        // Heapify the input array!
        MinHeap heap = new MinHeap(data);

        // Call remove-min K times!
        int k = 3;
        for (int i = 0; i < k; i++) {
            heap.delete_min();
        }
    }
}
