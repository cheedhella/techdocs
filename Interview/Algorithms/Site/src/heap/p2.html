public class PriorityQueueUsingHeap {
                
    public static void main(String[] args) {
        MaxHeap heap = new MaxHeap();
        heap.insert(9);
        heap.insert(3);
        heap.insert(15);
        heap.insert(1);
        heap.insert(12);
        heap.insert(5);
                
        while (!heap.isEmpty()) {
            heap.delete_max();
        }
    }
}
