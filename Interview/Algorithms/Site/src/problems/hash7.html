Problem: Find the first unique integer which appears only once in the whole array? 
            
import java.util.HashMap;
import java.util.Map;

public class NonUniqueApp {

    // Time Complexity: O(n);
    private static Integer findFirstUnique(int[] data) {
        Map dataSet = new HashMap(); // 
        for (int i = 0; i < data.length; i++) {
            if (dataSet.containsKey(data[i])) {
                int count = dataSet.get(data[i]);
                dataSet.put(data[i], count + 1);

            }
            else {
                dataSet.put(data[i], 1);
            }
        }

        return dataSet.entrySet().stream().filter(e -> e.getValue() == 1).map(Map.Entry::getKey)
                .findAny().orElse(null);
    }

    public static void main(String[] args) {
        int[] arr = { 2, 54, 7, 2, 6, 54 };
        System.out.println(findFirstUnique(arr));
    }
}

Output:
6
