Problem: Given a HashMap with all the points, can you find the starting point and print out the complete path from start to end? 
            

import java.util.HashMap;
import java.util.Map;

public class TracePathApp {

    private static String tracePath(Map points) {
        // Find starting point, a point which doesn't have route to!
        String startPoint = points.keySet().stream().filter(k -> !points.containsValue(k)).findAny()
                .orElse(null);

        // We could use ReverseMap for quick lookup by value!
        HashMap reverseMap = new HashMap<>();
        points.forEach((k, v) -> reverseMap.put(v, k));
        startPoint = points.keySet().stream().filter(k -> !reverseMap.containsKey(k)).findAny()
                .orElse(null);

        if (startPoint == null || startPoint.isEmpty()) {
            return "Couldn't find starting point!";
        }

        // Find path!
        String path = startPoint;
        String current = startPoint;
        while (points.get(current) != null) {
            path = path + " -> " + points.get(current);
            current = points.get(current);
        }
        return path;
    }

    public static void main(String[] args) {
        Map inputMap = new HashMap<>();
        inputMap.put("NewYork", "Chicago");
        inputMap.put("Boston", "Texas");
        inputMap.put("Missouri", "NewYork");
        inputMap.put("Texas", "Missouri");
        System.out.println("Path: " + tracePath(inputMap));
    }
}

Output:
Path: Boston -> Texas -> Missouri -> NewYork -> Chicago
