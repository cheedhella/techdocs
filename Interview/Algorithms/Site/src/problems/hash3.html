Problem: By definition, (a,b) and (c,d) are symmetric pairs if a = d and b = c; 
            

import java.util.HashMap;
import java.util.Map;

public class SymmetricApp {

    private static String findSymmetric(int[][] input) {
        String output = "";
        Map inputMap = new HashMap<>();
        for (int i = 0; i < input.length; i++) {
            // take current pair
            Integer a = input[i][0];
            Integer b = input[i][1];

            // check if symmetric pair exists in the map?
            Integer value = inputMap.get(b);
            if (value != null && value == a) {
                output += "{" + b + "," + a + "}";
            }
            else {
                inputMap.put(a, b);
            }
        }
        return output;
    }

    public static void main(String[] args) {
        int[][] input = { { 1, 2 }, { 3, 4 }, { 5, 9 }, { 4, 3 }, { 9, 5 } };
        System.out.println("Symmetric Pairs: " + findSymmetric(input));
    }
}

Output:
Symmetric Pairs: {3,4}{5,9}
