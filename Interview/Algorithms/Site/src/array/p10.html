• Given an integer array, move all 0s if any to the left while maintaining the order of other elements in the array;
        
import java.util.Arrays;

public class MoveZeros {

    private static void moveZerosToLeft(int[] nums) {
        int count = 0;
        for (int i = nums.length - 1; i >= 0; i--) {
            if (nums[i] == 0) {
                count++;
            }
            else {
                nums[i + count] = nums[i];
            }
        }
        for (int i = 0; i < count; i++) {
            nums[i] = 0;
        }
    }

    public static void main(String[] args) {
        int[] data = { 0, 1, 10, 20, 0, 59, 63, 0, 88 };
        System.out.println("Input: " + Arrays.toString(data));
        moveZerosToLeft(data);
        System.out.println("Output: " + Arrays.toString(data));
    }
}

// Output:
Input: [0, 1, 10, 20, 0, 59, 63, 0, 88]
Output: [0, 0, 0, 1, 10, 20, 59, 63, 88]
