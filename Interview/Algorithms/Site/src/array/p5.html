import java.util.Arrays;

public class P5 {
    private static void rightRotate(int[] data) {
        int lastEl = data[data.length - 1];
        for (int i = data.length - 1; i > 0; i--) {
            data[i] = data[i - 1];
        }
        data[0] = lastEl;
    }

    public static void main(String[] args) {
        int[] data = { 9, 2, 3, 6 };
        rightRotate(data);
        System.out.println("After right rotating: " + Arrays.toString(data));
    }
}

// Output:
After right rotating: [6, 9, 2, 3]
