• Given a String, sort all words in it by it's length and if word length is same, sort by alphabetical order;
        

import java.util.Collections;
import java.util.Comparator;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.Stream;

public class OrderingEg {
    public static void main(String args[]) {
        String input = "Mobile and Laptop are not exempt";
        List wordsList = Stream.of(input.split(" ")).map(String::trim)
                .collect(Collectors.toList());

        System.out.println("Before sorting: " + wordsList);
        Comparator comparator = (s1, s2) -> Integer.compare(s1.length(), s2.length());
        Collections.sort(wordsList, comparator);
        System.out.println("After sorting by length: " + wordsList);

        List wordsList2 = Stream.of(input.split(" ")).map(String::trim)
                .collect(Collectors.toList());

        System.out.println("Before sorting: " + wordsList2);
        Collections.sort(wordsList2, new Comparator() {
            @Override
            public int compare(String o1, String o2) {
                if (o1.length() != o2.length()) {
                    return o1.length() - o2.length();
                }
                else {
                    return o1.compareToIgnoreCase(o2);
                }
            }
        });

        System.out.println("After sorting by length and alphabetically: " + wordsList2);
    }
}

Output:
Before sorting: [Mobile, and, Laptop, are, not, exempt]
After sorting by length: [and, are, not, Mobile, Laptop, exempt]

Before sorting: [Mobile, and, Laptop, are, not, exempt]
After sorting by length and alphabetically: [and, are, not, exempt, Laptop, Mobile]
