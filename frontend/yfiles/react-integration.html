How do you integrate yFiles with React?
yFiles can be integrated into React applications using hooks and refs. 
import React, { useEffect, useRef } from 'react';
import {
    GraphComponent,
    GraphEditorInputMode,
    ShapeNodeStyle,
    HierarchicLayout
} from 'yfiles';

function GraphVisualization() {
    const divRef = useRef(null);
    const graphComponentRef = useRef(null);
    
    useEffect(() => {
        // Initialize GraphComponent
        const graphComponent = new GraphComponent(divRef.current);
        graphComponentRef.current = graphComponent;
        
        // Set up input mode
        graphComponent.inputMode = new GraphEditorInputMode();
        
        // Get graph
        const graph = graphComponent.graph;
        
        // Set default style
        graph.nodeDefaults.style = new ShapeNodeStyle({
            fill: 'lightblue',
            shape: 'round-rectangle'
        });
        
        // Create sample graph
        const node1 = graph.createNode();
        const node2 = graph.createNode();
        graph.addLabel(node1, 'Node 1');
        graph.addLabel(node2, 'Node 2');
        graph.createEdge(node1, node2);
        
        // Apply layout
        graphComponent.morphLayout(new HierarchicLayout());
        
        // Cleanup
        return () => {
            graphComponent.cleanUp();
        };
    }, []);
    
    return (
        

    );
}

export default GraphVisualization;
How do you pass data to the React component?
function GraphVisualization({ nodes, edges }) {
    const divRef = useRef(null);
    const graphComponentRef = useRef(null);
    
    useEffect(() => {
        const graphComponent = new GraphComponent(divRef.current);
        graphComponentRef.current = graphComponent;
        
        const graph = graphComponent.graph;
        
        // Build graph from data
        const nodeMap = new Map();
        
        nodes.forEach(nodeData => {
            const node = graph.createNode();
            graph.addLabel(node, nodeData.name);
            node.tag = nodeData;
            nodeMap.set(nodeData.id, node);
        });
        
        edges.forEach(edgeData => {
            const source = nodeMap.get(edgeData.from);
            const target = nodeMap.get(edgeData.to);
            if (source && target) {
                graph.createEdge(source, target);
            }
        });
        
        graphComponent.morphLayout(new HierarchicLayout());
        
        return () => graphComponent.cleanUp();
    }, [nodes, edges]);
    
    return 
;
}

const nodesData = [
    { id: 1, name: 'Alice' },
    { id: 2, name: 'Bob' }
];

    { from: 1, to: 2 }
