What is EdgeReconnectionPortCandidateProvider?
This provider controls which ports are valid targets when reconnecting edges, allowing you to implement custom connection rules and constraints.
How do you customize edge reconnection?
// Custom port candidate provider
class CustomPortCandidateProvider extends BaseClass(IEdgeReconnectionPortCandidateProvider) {
    getSourcePortCandidates(context, target) {
        const candidates = new List();
        const node = context.item.sourceNode;
        
        // Only allow reconnection from output ports
        node.ports.forEach(port => {
            if (port.tag?.type === 'output') {
                candidates.add(new DefaultPortCandidate(port));
            }
        });
        
        return candidates;
    }
    
    getTargetPortCandidates(context, source) {
        const candidates = new List();
        const node = context.item.targetNode;
        
        // Only allow reconnection to input ports
        node.ports.forEach(port => {
            if (port.tag?.type === 'input') {
                candidates.add(new DefaultPortCandidate(port));
            }
        });
        
        return candidates;
    }
}

// Register provider
const decorator = graph.decorator.edgeDecorator;
decorator.edgeReconnectionPortCandidateProviderDecorator.setFactory(
    edge => new CustomPortCandidateProvider()
);
