What is HandleProvider?
HandleProvider allows you to customize the resize handles that appear when nodes are selected, enabling custom resize behavior and constraints.
How do you customize handles?
// Custom handle provider
class CustomHandleProvider extends BaseClass(IHandleProvider) {
    getHandles(context) {
        const handles = new List();
        const node = context.item;
        const layout = node.layout;
        
        // Add custom handles at specific positions
        handles.add(new CustomHandle(node, HandlePositions.NORTH_EAST));
        handles.add(new CustomHandle(node, HandlePositions.SOUTH_WEST));
        
        return handles;
    }
}

// Register custom handle provider
const decorator = graph.decorator.nodeDecorator;
decorator.handleProviderDecorator.setFactory(
    node => new CustomHandleProvider()
);
