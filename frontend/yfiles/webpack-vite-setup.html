// vite.config.js
import { defineConfig } from 'vite';

export default defineConfig({
    optimizeDeps: {
        include: ['yfiles']
    },
    server: {
        port: 3000
    },
    build: {
        target: 'es2015',
        outDir: 'dist'
    }
});

How do you configure Webpack for yFiles?
// webpack.config.js
const path = require('path');

module.exports = {
    entry: './src/index.js',
    output: {
        filename: 'bundle.js',
        path: path.resolve(__dirname, 'dist')
    },
    module: {
        rules: [
            {
                test: /\.js$/,
                exclude: /node_modules/,
                use: {
                    loader: 'babel-loader',
                    options: {
                        presets: ['@babel/preset-env']
                    }
                }
            }
        ]
    },
    resolve: {
        extensions: ['.js']
    },
    devServer: {
        contentBase: './dist',
        port: 3000
    }
};
