Can you use CSS with yFiles?
Yes! yFiles for HTML supports CSS styling for SVG elements, allowing you to style graph elements using CSS classes and stylesheets.
How do you apply CSS classes to nodes?
// Create a style that uses CSS classes
class CssNodeStyle extends NodeStyleBase {
    createVisual(context, node) {
        const layout = node.layout;
        const g = document.createElementNS('http://www.w3.org/2000/svg', 'g');
        
        const rect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
        rect.setAttribute('width', layout.width);
        rect.setAttribute('height', layout.height);
        
        // Apply CSS class
        rect.setAttribute('class', node.tag?.cssClass || 'default-node');
        
        g.appendChild(rect);
        g.setAttribute('transform', `translate(${layout.x} ${layout.y})`);
        
        return new SvgVisual(g);
    }
    
    updateVisual(context, oldVisual, node) {
        return this.createVisual(context, node);
    }
}

// Apply CSS classes based on node type
graph.nodes.forEach(node => {
    let cssClass;
    switch (node.tag?.type) {
        case 'process':
            cssClass = 'process-node';
            break;
        case 'decision':
            cssClass = 'decision-node';
            break;
        case 'error':
            cssClass = 'error-node';
            break;
        default:
            cssClass = 'default-node';
    }
    node.tag = { ...node.tag, cssClass };
});
