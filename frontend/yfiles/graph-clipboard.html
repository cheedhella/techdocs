What is GraphClipboard?
GraphClipboard provides copy, cut, and paste functionality for graph elements, allowing users to duplicate and move parts of the graph.
How do you use clipboard operations?
const inputMode = new GraphEditorInputMode();

// Enable clipboard operations
inputMode.allowClipboardOperations = true;

// Copy selected elements
inputMode.keyboardInputMode.addKeyBinding(
    Key.C,
    ModifierKeys.CONTROL,
    () => {
        inputMode.copy();
        return true;
    }
);

// Cut selected elements
inputMode.keyboardInputMode.addKeyBinding(
    Key.X,
    ModifierKeys.CONTROL,
    () => {
        inputMode.cut();
        return true;
    }
);

// Paste
inputMode.keyboardInputMode.addKeyBinding(
    Key.V,
    ModifierKeys.CONTROL,
    () => {
        inputMode.paste();
        return true;
    }
);

graphComponent.inputMode = inputMode;
