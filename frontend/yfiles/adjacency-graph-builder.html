What is Adjacency Graph Builder?
Adjacency Graph Builder creates graphs from data where each node contains references to its neighbors (adjacency list format).
How do you use it?
// Data with adjacency information
const nodesData = [
    { id: 'A', name: 'Node A', neighbors: ['B', 'C'] },
    { id: 'B', name: 'Node B', neighbors: ['C', 'D'] },
    { id: 'C', name: 'Node C', neighbors: ['D'] },
    { id: 'D', name: 'Node D', neighbors: [] }
];

// Create builder
const builder = new AdjacencyGraphBuilder(graph);

// Configure nodes
const nodesSource = builder.createNodesSource({
    data: nodesData,
    id: 'id'
});

nodesSource.nodeCreator.createLabelBinding(data => data.name);

// Define adjacency (neighbors)
nodesSource.addSuccessorsSource(
    data => data.neighbors,
    nodesSource
);

// Build graph
builder.buildGraph();

// Apply layout
await graphComponent.morphLayout(new OrganicLayout());
