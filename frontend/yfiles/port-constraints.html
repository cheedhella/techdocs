What are Port Constraints?
Port Constraints control which edges can connect to which ports and from which directions, enabling precise control over connection topology.
How do you use port constraints?
// Create ports with constraints
const topPort = graph.addPort(
    node,
    FreeNodePortLocationModel.NODE_TOP_ANCHORED
);
topPort.tag = { type: 'input' };

const bottomPort = graph.addPort(
    node,
    FreeNodePortLocationModel.NODE_BOTTOM_ANCHORED
);
bottomPort.tag = { type: 'output' };

// Enforce constraints in layout
const layout = new HierarchicLayout();
const layoutData = new HierarchicLayoutData();

layoutData.sourcePortConstraints.mapper = edge => {
    return PortConstraint.create(PortSide.SOUTH);
};

layoutData.targetPortConstraints.mapper = edge => {
    return PortConstraint.create(PortSide.NORTH);
};

await graphComponent.morphLayout(layout, layoutData);
