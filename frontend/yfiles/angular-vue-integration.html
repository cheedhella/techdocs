How do you integrate yFiles with Angular?
// graph.component.ts
import { Component, OnInit, ElementRef, ViewChild } from '@angular/core';
import { GraphComponent, GraphEditorInputMode, HierarchicLayout } from 'yfiles';

@Component({
    selector: 'app-graph',
    template: '
'
})
export class GraphComponentWrapper implements OnInit {
    @ViewChild('graphDiv', { static: true }) graphDiv!: ElementRef;
    private graphComponent!: GraphComponent;
    
    ngOnInit() {
        this.graphComponent = new GraphComponent(this.graphDiv.nativeElement);
        this.graphComponent.inputMode = new GraphEditorInputMode();
        
        const graph = this.graphComponent.graph;
        
        // Create sample graph
        const node1 = graph.createNode();
        const node2 = graph.createNode();
        graph.addLabel(node1, 'Node 1');
        graph.addLabel(node2, 'Node 2');
        graph.createEdge(node1, node2);
        
        this.graphComponent.morphLayout(new HierarchicLayout());
    }
    
    ngOnDestroy() {
        this.graphComponent.cleanUp();
    }
}
