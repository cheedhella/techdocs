How do you set up yFiles with ES Modules?
yFiles for HTML is distributed as ES modules, allowing modern JavaScript import/export syntax. 
// Import yFiles modules
import {
    GraphComponent,
    GraphEditorInputMode,
    DefaultGraph,
    ShapeNodeStyle,
    PolylineEdgeStyle,
    HierarchicLayout
} from 'yfiles';

// Initialize GraphComponent
const graphComponent = new GraphComponent('#graphComponent');

// Set up input mode
graphComponent.inputMode = new GraphEditorInputMode();

// Get graph
const graph = graphComponent.graph;

// Set default styles
graph.nodeDefaults.style = new ShapeNodeStyle({
    fill: 'lightblue',
    shape: 'round-rectangle'
});

graph.edgeDefaults.style = new PolylineEdgeStyle({
    stroke: '2px gray'
});

// Create nodes and edges
const node1 = graph.createNode();
const node2 = graph.createNode();
graph.createEdge(node1, node2);

// Apply layout
await graphComponent.morphLayout(new HierarchicLayout());
