1. Getting Started

What is Zustand?
Zustand is a small, fast, and scalable state management solution for React. It uses a simplified flux principles with a hooks-based API.
  • Simple: Minimal boilerplate, easy to learn
  • Fast: Optimized for performance
  • Scalable: Works for small and large apps
  • No Context Provider: No wrapper needed
  • TypeScript First: Excellent TypeScript support
Why Zustand?
Advantages over useState:
  • No prop drilling - access state from any component
  • Centralized state management
  • Better for complex state logic
Advantages over Redux:
  • Much less boilerplate
  • No actions, reducers, or dispatch needed
  • Simpler API
  • Smaller bundle size (~1KB)
Advantages over Context API:
  • Better performance (no unnecessary re-renders)
  • No provider hell
  • Can use outside React components

Reading State
// Select entire state (causes re-render on any change)

// Select specific property (only re-renders when count changes)

// Select multiple properties

Updating State
const useStore = create((set) => ({
  count: 0,
  user: { name: 'John', age: 30 },
  
  // Simple update
  setCount: (count) => set({ count }),
  
  // Update based on previous state
  increment: () => set((state) => ({ count: state.count + 1 })),
  
  // Update nested state (immutably)
  updateName: (name) => set((state) => ({
    user: { ...state.user, name }
  })),
  
  // Update multiple properties
  updateUser: (name, age) => set((state) => ({
    user: { ...state.user, name, age }
  })),
  
  // Replace entire state
  reset: () => set({ count: 0, user: { name: '', age: 0 } })
}));

2. Core Concepts

Selectors
Selectors extract specific parts of state and prevent unnecessary re-renders: 
const useStore = create((set) => ({
  firstName: 'John',
  lastName: 'Doe',
  age: 30
}));

// Component only re-renders when firstName changes
function FirstName() {
  const firstName = useStore((state) => state.firstName);
  return 
{firstName}
;
}

// Computed selector
function FullName() {
  const fullName = useStore((state) => 
    `${state.firstName} ${state.lastName}`
  );
  return 
{fullName}
;
}

// Multiple selectors
function UserInfo() {
  const firstName = useStore((state) => state.firstName);
  const age = useStore((state) => state.age);
  return 
{firstName} - {age}
;
}
Actions
Actions are functions that update state: 
const useStore = create((set, get) => ({
  count: 0,
  
  // Simple action
  increment: () => set((state) => ({ count: state.count + 1 })),
  
  // Action using get() to read current state
  incrementBy: (amount) => {
    const currentCount = get().count;
    set({ count: currentCount + amount });
  },
  
  // Action calling another action
  incrementTwice: () => {
    get().increment();
    get().increment();
  },
  
  // Action with logic
  incrementIfLessThan10: () => {
    const count = get().count;
    if (count < 10) {
      set({ count: count + 1 });
    }
  }
}));
Immutability
Always create new objects when updating state: 
const useStore = create((set) => ({
  user: { name: 'John', age: 30, address: { city: 'NYC' } },
  
  // ❌ WRONG - Mutating state directly
  updateNameWrong: (name) => set((state) => {
    state.user.name = name;  // Don't do this!
    return state;
  }),
  
  // ✅ CORRECT - Creating new object
  updateName: (name) => set((state) => ({
    user: { ...state.user, name }
  })),
  
  // ✅ CORRECT - Nested object update
  updateCity: (city) => set((state) => ({
    user: {
      ...state.user,
      address: { ...state.user.address, city }
    }
  })),
  
  // ✅ CORRECT - Array update
  todos: [],
  addTodo: (todo) => set((state) => ({
    todos: [...state.todos, todo]
  })),
  removeTodo: (id) => set((state) => ({
    todos: state.todos.filter(t => t.id !== id)
  }))
}));
get() and set() API
const useStore = create((set, get) => ({
  count: 0,
  multiplier: 2,
  
  // set() - Update state
  increment: () => set((state) => ({ count: state.count + 1 })),
  
  // get() - Read current state
  getCount: () => get().count,
  
  // Using both
  multiplyCount: () => {
    const current = get().count;
    const multiplier = get().multiplier;
    set({ count: current * multiplier });
  },
  
  // Partial updates (Zustand merges automatically)
  updateMultiplier: (multiplier) => set({ multiplier })
}));

// Outside component access
const currentCount = useStore.getState().count;
useStore.setState({ count: 10 });
Subscribe API
Subscribe to state changes outside React: 
const useStore = create((set) => ({

// Subscribe to all changes

// Subscribe to specific property

  (state) => state.count,

    console.log('Count changed:', count);

// Cleanup

// Example: Sync with localStorage

3. Async & Side Effects

Async Actions
Zustand handles async actions naturally: 
const useStore = create((set, get) => ({
  user: null,
  loading: false,
  error: null,
  
  // Async action
  fetchUser: async (id) => {
    set({ loading: true, error: null });
    
    try {
      const response = await fetch(`/api/users/${id}`);
      const user = await response.json();
      set({ user, loading: false });
    } catch (error) {
      set({ error: error.message, loading: false });
    }
  },
  
  // Async action with cleanup
  fetchUserWithAbort: async (id) => {
    const controller = new AbortController();
    set({ loading: true, error: null });
    
    try {
      const response = await fetch(`/api/users/${id}`, {
        signal: controller.signal
      });
      const user = await response.json();
      set({ user, loading: false });
    } catch (error) {
      if (error.name !== 'AbortError') {
        set({ error: error.message, loading: false });
      }
    }
    
    return () => controller.abort();
  }
}));

immer - Mutable Updates
Write "mutable" code that produces immutable updates: 
import { create } from 'zustand';
import { immer } from 'zustand/middleware/immer';

const useStore = create(
  immer((set) => ({
    user: {
      name: 'John',
      address: {
        city: 'NYC',
        zip: '10001'
      }
    },
    
    // With immer, you can "mutate" state
    updateCity: (city) => set((state) => {
      state.user.address.city = city;
    }),
    
    // Array operations
    todos: [],
    addTodo: (todo) => set((state) => {
      state.todos.push(todo);
    }),
    removeTodo: (id) => set((state) => {
      const index = state.todos.findIndex(t => t.id === id);
      if (index !== -1) state.todos.splice(index, 1);
    })
  }))
);

// Without immer (manual immutability)
const useStore = create((set) => ({
  user: { name: 'John', address: { city: 'NYC' } },
  updateCity: (city) => set((state) => ({
    user: {
      ...state.user,
      address: { ...state.user.address, city }
    }
  }))
}));

Combine multiple middleware: 
import { create } from 'zustand';

6. Performance Optimization

Preventing Unnecessary Re-renders
// ❌ BAD - Re-renders on any state change
function Component() {
  const state = useStore();
  return 
{state.count}
;
}

// ✅ GOOD - Only re-renders when count changes
function Component() {
  const count = useStore((state) => state.count);
  return 
{count}
;
}

// ❌ BAD - Creates new object every time, always re-renders
function Component() {
  const data = useStore((state) => ({
    count: state.count,
    name: state.name
  }));
  return 
{data.count}
;
}

// ✅ GOOD - Use shallow comparison
import { shallow } from 'zustand/shallow';

function Component() {
  const { count, name } = useStore(
    (state) => ({ count: state.count, name: state.name }),
    shallow
  );
  return 
{count}
;
}

// ✅ GOOD - Select actions separately (they don't change)
function Component() {
  const count = useStore((state) => state.count);
  const increment = useStore((state) => state.increment);

{count}

type SortOrder = 'asc' | 'desc';

interface FilterState {
  products: Product[];
  searchQuery: string;
  selectedCategories: string[];
  priceRange: [number, number];
  showInStockOnly: boolean;
  sortField: SortField;
  sortOrder: SortOrder;
  
  setSearchQuery: (query: string) => void;
  toggleCategory: (category: string) => void;
  setPriceRange: (range: [number, number]) => void;
  setShowInStockOnly: (show: boolean) => void;
  setSorting: (field: SortField, order: SortOrder) => void;
  getFilteredProducts: () => Product[];
  resetFilters: () => void;
}

export const useFilterStore = create((set, get) => ({
  products: [],
  searchQuery: '',
  selectedCategories: [],
  priceRange: [0, 1000],
  showInStockOnly: false,
  sortField: 'name',
  sortOrder: 'asc',
  
  setSearchQuery: (query) => set({ searchQuery: query }),
  
  toggleCategory: (category) => set((state) => {
    const categories = state.selectedCategories;
    if (categories.includes(category)) {
      return { selectedCategories: categories.filter(c => c !== category) };
    }
    return { selectedCategories: [...categories, category] };
  }),
  
  setPriceRange: (range) => set({ priceRange: range }),
  
  setShowInStockOnly: (show) => set({ showInStockOnly: show }),
  
  setSorting: (field, order) => set({ sortField: field, sortOrder: order }),
  
  getFilteredProducts: () => {
    const state = get();
    let filtered = state.products;
    
    // Search filter
    if (state.searchQuery) {
      filtered = filtered.filter(p =>
        p.name.toLowerCase().includes(state.searchQuery.toLowerCase())
      );
    }
    
    // Category filter
    if (state.selectedCategories.length > 0) {
      filtered = filtered.filter(p =>
        state.selectedCategories.includes(p.category)
      );
    }
    
    // Price range filter
    filtered = filtered.filter(p =>
      p.price >= state.priceRange[0] && p.price <= state.priceRange[1]
    );
    
    // Stock filter
    if (state.showInStockOnly) {
      filtered = filtered.filter(p => p.inStock);
    }
    
    // Sorting
    filtered.sort((a, b) => {
      const field = state.sortField;
      const order = state.sortOrder === 'asc' ? 1 : -1;
      return a[field] > b[field] ? order : -order;
    });
    
    return filtered;
  },
  
  resetFilters: () => set({
    searchQuery: '',
    selectedCategories: [],
    priceRange: [0, 1000],
    showInStockOnly: false,

Comparison with Other Libraries
Zustand vs Redux:
  • Zustand: ~1KB, Redux: ~3KB + React-Redux ~5KB
  • Zustand: No boilerplate, Redux: Actions, reducers, types
  • Zustand: Direct mutations with immer, Redux: Requires Redux Toolkit
Zustand vs Context API:
  • Zustand: No provider needed, Context: Requires Provider wrapper
  • Zustand: Better performance, Context: Can cause unnecessary re-renders
  • Zustand: Can use outside React, Context: React-only
Zustand vs Jotai:
  • Zustand: Store-based, Jotai: Atom-based
  • Zustand: Simpler for global state, Jotai: Better for derived state
  • Zustand: Easier to debug, Jotai: More composable
Zustand vs Valtio:
  • Zustand: Immutable updates, Valtio: Mutable proxy-based
  • Zustand: Explicit selectors, Valtio: Automatic tracking
  • Zustand: More control, Valtio: More magic

Best Practices
  • Keep stores focused: One store per domain (user, cart, settings)
  • Use selectors: Always select only what you need
  • Avoid storing derived state: Compute on the fly or use getters
  • Use TypeScript: Type safety prevents bugs
  • Persist carefully: Don't persist sensitive data or temporary state
  • Use middleware wisely: Only add what you need
  • Test your stores: Stores are easy to unit test
  • Keep actions simple: Complex logic should be in services
  • Use shallow comparison: When selecting multiple values
  • Don't abuse get(): Prefer using state in set() callback

Error Handling Patterns
const useStore = create((set) => ({
  data: null,
  loading: false,
  error: null,
  
  // Pattern 1: Try-catch with error state
  fetchData: async () => {
    set({ loading: true, error: null });
    try {
      const response = await fetch('/api/data');
      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }
      const data = await response.json();
      set({ data, loading: false });
    } catch (error) {
      set({
        error: error instanceof Error ? error.message : 'Unknown error',
        loading: false
      });
    }
  },
  
  // Pattern 2: Return error instead of storing
  fetchDataWithReturn: async () => {
    set({ loading: true });
    try {
      const response = await fetch('/api/data');
      const data = await response.json();
      set({ data, loading: false });
      return { success: true, data };
    } catch (error) {
      set({ loading: false });
      return { success: false, error: error.message };
    }
  },
  
  // Pattern 3: Error boundary integration
  fetchDataWithThrow: async () => {
    set({ loading: true });
    try {
      const response = await fetch('/api/data');
      const data = await response.json();
      set({ data, loading: false });
    } catch (error) {
      set({ loading: false });
      throw error;  // Let error boundary handle it
    }
  }
}));

// store.test.ts
import { renderHook, act } from '@testing-library/react';
import { useStore } from './store';

describe('useStore', () => {
  beforeEach(() => {
    // Reset store before each test
    useStore.setState({ count: 0 });
  });
  
  it('should have initial count of 0', () => {
    const { result } = renderHook(() => useStore());
    expect(result.current.count).toBe(0);
  });
  
  it('should increment count', () => {
    const { result } = renderHook(() => useStore());
    
    act(() => {
      result.current.increment();
    });
    
    expect(result.current.count).toBe(1);
  });
  
  it('should decrement count', () => {
    const { result } = renderHook(() => useStore());
    
    act(() => {
      result.current.increment();
      result.current.decrement();
    });
    
    expect(result.current.count).toBe(0);
  });
  
  it('should reset count', () => {
    const { result } = renderHook(() => useStore());
    
    act(() => {
      result.current.increment();
      result.current.increment();
      result.current.reset();
    });
    
    expect(result.current.count).toBe(0);
  });
});
