1. What is Undo and Redo?
  • Undo - Reverting the most recent change;
  • Redo - Reversing undo, restoring an action that you just undid;
  • If your application need only Undo, then single stack is enough;
  • If your application need only Undo and Redo, then it can be implemented using either 2 stacks OR 1 Deque;
Undo using Stack

import java.util.Stack;

public class SimpleUndo {
    private Stack actions = new Stack<>();
    private StringBuilder text = new StringBuilder();

    // Perform a typing action
    public void type(String word) {
        actions.push(text.toString());  // Save current state before change
        text.append(word);
        System.out.println("Typed: " + word);
    }

    // Undo the last action
    public void undo() {
        if (!actions.isEmpty()) {
            text = new StringBuilder(actions.pop() + " ");  // Restore previous state
            System.out.println("Undo performed.");
        } else {
            System.out.println("Nothing to undo!");
        }
    }

    // Display current text
    public void show() {
        System.out.println("Current text: " + text);
    }

    public static void main(String[] args) {
        SimpleUndo editor = new SimpleUndo();

        editor.type("one");
        editor.type("two");
        editor.type("three");
        editor.show();

        editor.undo();
        editor.show();

        editor.undo();
        editor.show();

        editor.undo();
        editor.show();

        editor.undo();  // Nothing to undo
    }
}

Output:
Typed: one 
Typed: two
Typed: three
Current text: one two three
Undo performed.
Current text: one two 
Undo performed.
Current text: one
Undo performed.
Current text: 
Nothing to undo!


Undo-Redo using 2 Stacks

import java.util.Stack;

public class UndoRedoUsingStacks {
    private Stack undoStack = new Stack<>();
    private Stack redoStack = new Stack<>();

    public void doAction(String action) {
        undoStack.push(action);
        redoStack.clear(); // Once a new action happens, redo history resets
        System.out.println("Performed: " + action);
    }

    public void undo() {
        if (!undoStack.isEmpty()) {
            String action = undoStack.pop();
            redoStack.push(action);
            System.out.println("Undo: " + action);
        } else {
            System.out.println("Nothing to undo");
        }
    }

    public void redo() {
        if (!redoStack.isEmpty()) {
            String action = redoStack.pop();
            undoStack.push(action);
            System.out.println("Redo: " + action);
        } else {
            System.out.println("Nothing to redo");
        }
    }

    public static void main(String[] args) {
        UndoRedoUsingStacks editor = new UndoRedoUsingStacks();
        editor.doAction("Type A");
        editor.doAction("Type B");
        editor.undo();
        editor.redo();
    }
}

Output:
Performed: Type A
Performed: Type B
Undo: Type B
Redo: Type B


Undo-Redo using Deque

import java.util.Deque;
import java.util.LinkedList;

public class UndoRedoUsingDeque {
    private Deque undoDeque = new LinkedList<>();
    private Deque redoDeque = new LinkedList<>();

    public void doAction(String action) {
        undoDeque.push(action);    // Add to top
        redoDeque.clear();         // Clear redo history
        System.out.println("Performed: " + action);
    }

    public void undo() {
        if (!undoDeque.isEmpty()) {
            String action = undoDeque.pop();
            redoDeque.push(action);
            System.out.println("Undo: " + action);
        } else {
            System.out.println("Nothing to undo");
        }
    }

    public void redo() {
        if (!redoDeque.isEmpty()) {
            String action = redoDeque.pop();
            undoDeque.push(action);
            System.out.println("Redo: " + action);
        } else {
            System.out.println("Nothing to redo");
        }
    }

    public static void main(String[] args) {
        UndoRedoUsingDeque editor = new UndoRedoUsingDeque();
        editor.doAction("Draw Circle");
        editor.doAction("Draw Square");
        editor.undo();
        editor.redo();
    }
}

Output:
Performed: Draw Circle
Performed: Draw Square
Undo: Draw Square
Redo: Draw Square
