Maximum Sum Subarray of Size K

Problem: Given an array of positive numbers and a positive number K, find the maximum sum of any contiguous subarray of size K.

Algorithm: Sliding Window

  • Brute Force Approach: Iterate through each possible subarray of size K and calculate its sum. This takes O(N*K) time.
  • Sliding Window Approach:
    • Instead of re-calculating the sum of each subarray, we can reuse the sum of the previous subarray.
    • When we move the window one step forward, we add the next element and subtract the element that is going out of the window.
    • This reduces the time complexity to O(N).

Complexity:

  • Time Complexity: O(N)
  • Space Complexity: O(1)

public class MaxSumSubArrayOfSizeK {
    // Time Complexity: O(N*K);
    public static int findMaxSumSubArray(int k, int[] arr) {
        int maxSum = 0;
        for (int i = 0; i <= arr.length - k; i++) {
            int windowSum = 0;
            for (int j = i; j < i + k; j++) {
                windowSum += arr[j];
            }
            maxSum = Math.max(maxSum, windowSum);
        }
        return maxSum;
    }

    // Time Complexity: O(N);
    public static int findMaxSumSubArray2(int k, int[] arr) {
        int windowSum = 0;
        for (int i = 0; i < k; i++) {
            windowSum += arr[i];
        }

        int maxSum = windowSum, windowStart = 0;
        for (int i = k; i < arr.length; i++) {
            windowSum += arr[i]; // add the next element
            windowSum -= arr[windowStart]; // subtract the element going out
            windowStart++; // slide the window ahead
            maxSum = (windowSum > maxSum) ? windowSum : maxSum;
        }

        return maxSum;
    }

    public static void main(String[] args) {
        int maxSum1 = MaxSumSubArrayOfSizeK.findMaxSumSubArray(3, new int[] { 2, 1, 5, 1, 3, 2 });
        System.out.println("Maximum sum of a subarray of size K: " + maxSum1);
        int maxSum2 = MaxSumSubArrayOfSizeK.findMaxSumSubArray(2, new int[] { 2, 3, 4, 1, 5 });
        System.out.println("Maximum sum of a subarray of size K: " + maxSum2);
        int maxSum3 = MaxSumSubArrayOfSizeK.findMaxSumSubArray2(3, new int[] { 2, 1, 5, 1, 3, 2 });
        System.out.println("Maximum sum of a subarray of size K: " + maxSum3);
        int maxSum4 = MaxSumSubArrayOfSizeK.findMaxSumSubArray2(2, new int[] { 2, 3, 4, 1, 5 });
        System.out.println("Maximum sum of a subarray of size K: " + maxSum4);
    }
}
