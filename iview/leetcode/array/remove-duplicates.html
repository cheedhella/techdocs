How to Remove duplicate elements from an Array?

import java.util.Arrays;
import java.util.HashSet;
import java.util.LinkedHashSet;
import java.util.Set;
import java.util.stream.Collectors;

public class RemoveDuplicates {
    public static void main(String[] args) {
        Integer[] numbers = {1, 2, 3, 2, 1, 4, 5, 4};

        // Method 1: Using LinkedHashSet (Preserves Order)
        Set<Integer> set = new LinkedHashSet<>(Arrays.asList(numbers));
        System.out.println("Using LinkedHashSet: " + set);

        // Method 2: Using Java 8 Streams
        Integer[] uniqueNumbers = Arrays.stream(numbers)
                                        .distinct()
                                        .toArray(Integer[]::new);
        System.out.println("Using Streams: " + Arrays.toString(uniqueNumbers));
    }
}

Output:
Using LinkedHashSet: [1, 2, 3, 4, 5]
Using Streams: [1, 2, 3, 4, 5]
