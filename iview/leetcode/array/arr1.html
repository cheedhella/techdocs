Given an array of integers, remove even numbers from it?

import java.util.Arrays;
import java.util.stream.IntStream;

public class P1 {
    public static void main(String[] args) {
        int[] a1 = { 1, 2, 4, 5, 10, 6, 3 };
        int[] evenInts = IntStream.of(a1).filter(n -> n % 2 == 0).toArray();
        System.out.println("Even numbers removed: " + Arrays.toString(evenInts));
    }
}

Output:
Even numbers removed: [2, 4, 10, 6]
