1. What is Sliding Window Maximum (Maximum of all subarrays of size k)
  • Given an array and an integer K, find the maximum for each and every contiguous subarray of size k;
  • Take a Deque, dq;
  • For each element in the first k elements
  • if dq is empty, insert;
  • else
  • Remove all the elements at the end of dq, those are smaller than the current element;
  • Insert current element at the end of dq;
  • Print the maximum element of the first window;
  • For each element from k to the end of the array
  • Print the front element of the deque;
  • Remove all the elements from the front of the queue if they are out of the current window;
  • Remove all the elements at the end of dq, those are smaller than the current element;
  • Insert current element at the end of dq;
  • Print the maximum element of the last window;
Palindrome Check with ArrayDeque

import java.util.*;

public class SlidingWindowMaximum {
    public static int[] maxSlidingWindow(int[] nums, int k) {
        if (nums == null || k <= 0) return new int[0];

        int n = nums.length;
        int[] result = new int[n - k + 1];
        Deque deque = new ArrayDeque<>(); // stores indices

        for (int i = 0; i < n; i++) {
            // Remove indices out of the current window
            if (!deque.isEmpty() && deque.peekFirst() <= i - k) {
                deque.pollFirst();
            }

            // Remove smaller elements from the back (they’re useless)
            while (!deque.isEmpty() && nums[deque.peekLast()] < nums[i]) {
                deque.pollLast();
            }

            // Add current index
            deque.offerLast(i);

            // Store the max value once the first window is complete
            if (i >= k - 1) {
                result[i - k + 1] = nums[deque.peekFirst()];
            }
        }

        return result;
    }

    public static void main(String[] args) {
        int[] nums = {1, 3, -1, -3, 5, 3, 6, 7};
        int k = 3;
        int[] result = maxSlidingWindow(nums, k);

        System.out.println("Sliding window maximums: " + Arrays.toString(result));
    }
}


Output:
Sliding window maximums: [3, 3, 5, 5, 6, 7]
