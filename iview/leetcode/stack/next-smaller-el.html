Next Smaller Element (Classical Stack Problem)

Given an array of positive integers, for each array element output the nearest smaller element, i.e., the first smaller element that comes after the element in the array. If no such element exists, output -1.

Example:
Input: [3, 2, 4, 5, 1, 6]
Output: [2, 1, 1, 1, -1, -1]

Approach 1: Brute Force (O(N^2) Time, O(1) Space)
The simplest approach is to use two nested loops. For every element at index i, we iterate from j = i + 1 to the end of the array to find the first element that is smaller than arr[i]. If found, we store it; if we reach the end without finding any, we store -1. This results in a Time Limit Exceeded (TLE) for large inputs.

Approach 2: Monotonic Stack (O(N) Time, O(N) Space)
Using a Monotonic Stack. We iterate through the array backwards (from right to left). For each element, we pop elements from the stack that are greater than or equal to the current element. The top of the stack will then be the nearest smaller element. Finally, we push the current element onto the stack. This gives us an O(N) time complexity.

import java.util.Arrays;
import java.util.Stack;

public class NextSmallerElement {
    
    // Approach 1: Brute Force O(N^2)
    public int[] nextSmallerElementBruteForce(int[] arr) {
        int n = arr.length;
        int[] result = new int[n];
        
        for (int i = 0; i < n; i++) {
            int current = arr[i];
            boolean found = false;
            for (int j = i + 1; j < n; j++) {
                if (arr[j] < current) {
                    result[i] = arr[j];
                    found = true;
                    break;
                }
            }
            if (!found) {
                result[i] = -1;
            }
        }
        return result;
    }

    // Approach 2: Monotonic Stack O(N)
    public int[] nextSmallerElement(int[] arr) {
        int n = arr.length;
        int[] result = new int[n];
        Stack<Integer> stack = new Stack<>();

        // Traverse from right to left
        for (int i = n - 1; i >= 0; i--) {
            // Remove all elements from stack that are greater than or equal to current element
            // because they can never be the "Next Smaller Element" for any element to the left
            while (!stack.isEmpty() && stack.peek() >= arr[i]) {
                stack.pop();
            }

            // If stack is empty, there is no smaller element to the right
            if (stack.isEmpty()) {
                result[i] = -1;
            } else {
                // The top element is the nearest smaller element
                result[i] = stack.peek();
            }

            // Push current element to stack for future elements to compare against
            stack.push(arr[i]);
        }
        
        return result;
    }

    public static void main(String[] args) {
        NextSmallerElement solver = new NextSmallerElement();
        
        int[] arr1 = {3, 2, 4, 5, 1, 6};
        System.out.println("Input:  " + Arrays.toString(arr1));
        System.out.println("Output (Brute Force): " + Arrays.toString(solver.nextSmallerElementBruteForce(arr1)));
        System.out.println("Output (Stack):       " + Arrays.toString(solver.nextSmallerElement(arr1)));
        
        System.out.println();
        
        int[] arr2 = {3, 1, 2};
        System.out.println("Input:  " + Arrays.toString(arr2));
        System.out.println("Output (Brute Force): " + Arrays.toString(solver.nextSmallerElementBruteForce(arr2)));
        System.out.println("Output (Stack):       " + Arrays.toString(solver.nextSmallerElement(arr2)));
    }
}
