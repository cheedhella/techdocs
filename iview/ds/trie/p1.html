• Autocomplete system works by descending the tree as far as the text you'd feed it goes and then the system would carry on with the tree traversal to suggest you the possible completions;
  • Imagine you are designing a user signup form, which has a country field with lot of options; We can create a trie with list of all the countries; When the users type part of their country, we can lookup the trie and show all available options to the user;
        
import java.util.List;

public class LookAheadApp {

    public List suggest(String prefix) {
        List list = new ArrayList<>();
        Node current = root;
        StringBuffer curr = new StringBuffer();
        for (char c : prefix.toCharArray()) {
            current = current.getChildren().get(c);
            if (current == null)
                return list;
            curr.append(c);
        }
        suggestHelper(current, list, new String(curr));
        return list;
    }
  
    private void suggestHelper(Node root, List list, String curr) {
        if (root.isWord()) {
            list.add(curr.toString());
        }
  
        if (root.getChildren() == null || root.getChildren().isEmpty())
            return;
  
        root.getChildren().forEach((ch, node) -> {
            String newCurr = curr + Character.toString(ch);
            suggestHelper(node, list, newCurr);
        });
    }
    
    public static void main(String[] args) {
        Trie d1 = getTrie();
        List suggestions = d1.suggest("th");
        suggestions.forEach(s -> System.out.println(s));
    }

    private static Trie getTrie() {
        Trie d1 = new Trie();
        d1.insert("the");
        d1.insert("a");
        d1.insert("there");
        d1.insert("answer");
        d1.insert("any");
        d1.insert("by");
        d1.insert("bye");
        d1.insert("their");
        return d1;
    }
}

// Output:
the
there
their
