Step 4: Create REST Controller

HelloController.java

package com.example.demo;

import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import java.time.LocalDateTime;
import java.util.HashMap;
import java.util.Map;

@RestController
@RequestMapping("/api")
public class HelloController {
    
    @GetMapping("/hello")
    public Map<String, Object> hello() {
        Map<String, Object> response = new HashMap<>();
        response.put("message", "Hello from Spring Boot with Jib!");
        response.put("timestamp", LocalDateTime.now());
        response.put("status", "success");
        return response;
    }
    
    @GetMapping("/hello/{name}")
    public Map<String, Object> helloName(@PathVariable String name) {
        Map<String, Object> response = new HashMap<>();
        response.put("message", "Hello, " + name + "!");
        response.put("timestamp", LocalDateTime.now());
        response.put("status", "success");
        return response;
    }
    
    @GetMapping("/health")
    public Map<String, String> health() {
        Map<String, String> response = new HashMap<>();
        response.put("status", "UP");
        response.put("application", "Spring Boot Jib Demo");
        return response;
    }
}

Step 7: Build Docker Image with Jib

Option 1: Build to Local Docker Daemon

# Build image to local Docker
mvn compile jib:dockerBuild

# Verify the image
docker images | grep spring-boot-jib-demo

# Run the container
docker run -p 8080:8080 spring-boot-jib-demo:1.0-SNAPSHOT

# Test the containerized app
curl http://localhost:8080/api/hello

Option 2: Build and Push to Docker Hub

# Login to Docker Hub
docker login

# Build and push (replace YOUR_USERNAME with your Docker Hub username)
mvn compile jib:build -Dimage=docker.io/YOUR_USERNAME/spring-boot-jib-demo:latest

# Or use command-line authentication
mvn compile jib:build \
  -Djib.to.auth.username=YOUR_USERNAME \
  -Djib.to.auth.password=YOUR_PASSWORD

Option 3: Build as Tarball

# Build as tarball
mvn compile jib:buildTar

# Load the tarball into Docker
docker load --input target/jib-image.tar

# Run the container
docker run -p 8080:8080 spring-boot-jib-demo:1.0-SNAPSHOT
Step 8: Run and Test the Container

Run with Docker:

# Run in foreground
docker run -p 8080:8080 spring-boot-jib-demo:1.0-SNAPSHOT

# Run in background (detached mode)
docker run -d -p 8080:8080 --name spring-app spring-boot-jib-demo:1.0-SNAPSHOT

# Check logs
docker logs spring-app

# Follow logs
docker logs -f spring-app

# Stop the container
docker stop spring-app

# Remove the container
docker rm spring-app

Test All Endpoints:

# Test hello endpoint
curl http://localhost:8080/api/hello

# Test hello with name
curl http://localhost:8080/api/hello/Docker

# Test health endpoint
curl http://localhost:8080/api/health

# Test actuator health
curl http://localhost:8080/actuator/health
Step 9: Advanced Jib Configuration

Production-Ready Configuration:

<plugin>
    <groupId>com.google.cloud.tools</groupId>
    <artifactId>jib-maven-plugin</artifactId>
    <version>3.4.0</version>
    <configuration>
        <from>
            <image>gcr.io/distroless/java17-debian12:nonroot</image>
        </from>
        <to>
            <image>docker.io/YOUR_USERNAME/spring-boot-jib-demo</image>
            <tags>
                <tag>latest</tag>
                <tag>${project.version}</tag>
            </tags>
        </to>
        <container>
            <jvmFlags>
                <jvmFlag>-Xms256m</jvmFlag>
                <jvmFlag>-Xmx512m</jvmFlag>
                <jvmFlag>-XX:+UseContainerSupport</jvmFlag>
                <jvmFlag>-XX:MaxRAMPercentage=75.0</jvmFlag>
                <jvmFlag>-Djava.security.egd=file:/dev/./urandom</jvmFlag>
            </jvmFlags>
            <ports>
                <port>8080</port>
            </ports>
            <environment>
                <SPRING_PROFILES_ACTIVE>production</SPRING_PROFILES_ACTIVE>
                <SPRING_OUTPUT_ANSI_ENABLED>ALWAYS</SPRING_OUTPUT_ANSI_ENABLED>
            </environment>
            <labels>
                <maintainer>your-email@example.com</maintainer>
                <version>${project.version}</version>
                <description>Spring Boot Demo with Jib</description>
            </labels>
            <user>nonroot:nonroot</user>
            <format>OCI</format>
            <creationTime>USE_CURRENT_TIMESTAMP</creationTime>
        </container>
    </configuration>
</plugin>
Step 10: Docker Compose Setup (Optional)

Create docker-compose.yml for easy deployment:

version: '3.8'

services:
  spring-app:
    image: spring-boot-jib-demo:1.0-SNAPSHOT
    container_name: spring-boot-app
    ports:
      - "8080:8080"
    environment:
      - SPRING_PROFILES_ACTIVE=docker
      - JAVA_OPTS=-Xms256m -Xmx512m
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8080/actuator/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 40s
    restart: unless-stopped
    networks:
      - app-network

networks:
  app-network:
    driver: bridge

Run with Docker Compose:

# Start the application
docker-compose up -d

# View logs
docker-compose logs -f

# Stop the application
docker-compose down

# Rebuild and restart
docker-compose up -d --build

Troubleshooting

Common Issues and Solutions:

1. Authentication Failed:

# Login to Docker Hub first
docker login

# Or use command-line auth
mvn jib:build -Djib.to.auth.username=USERNAME -Djib.to.auth.password=PASSWORD

2. Port Already in Use:

# Use a different port
docker run -p 9090:8080 spring-boot-jib-demo:1.0-SNAPSHOT

# Or stop the conflicting container
docker ps
docker stop <container-id>

3. Out of Memory:

# Increase JVM memory
docker run -p 8080:8080 \
  -e JAVA_OPTS="-Xms512m -Xmx1024m" \
  spring-boot-jib-demo:1.0-SNAPSHOT

4. Enable Debug Logging:

# Build with verbose output
mvn compile jib:build -X

# Run container with debug logging
docker run -p 8080:8080 \
  -e LOGGING_LEVEL_ROOT=DEBUG \
  spring-boot-jib-demo:1.0-SNAPSHOT
Key Benefits of Using Jib
  • No Docker Required: Build images without Docker daemon installed
  • Fast Builds: Intelligent layer caching - only changed layers are rebuilt
  • Reproducible: Same source code always produces the same image
  • Optimized Layers: Dependencies, resources, and classes in separate layers
  • Secure by Default: Uses distroless images and runs as non-root
  • Maven Integration: Seamlessly integrates with Maven build lifecycle
  • No Dockerfile: No need to learn Dockerfile syntax or best practices
