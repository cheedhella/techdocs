5. Extra Directories Configuration

Adding Additional Files and Directories to Your Image
<configuration>
    <extraDirectories>
        <!-- Simple path -->
        <paths>
            <path>src/main/jib</path>
        </paths>
        
        <!-- Advanced: Multiple paths with different targets -->
        <paths>
            <path>
                <from>src/main/custom-files</from>
                <into>/app/config</into>
            </path>
            <path>
                <from>scripts</from>
                <into>/app/scripts</into>
                <includes>*.sh</includes>
                <excludes>test-*.sh</excludes>
            </path>
        </paths>
        
        <!-- Set permissions for files -->
        <permissions>
            <permission>
                <file>/app/scripts/startup.sh</file>
                <mode>755</mode>
            </permission>
            <permission>
                <file>/app/config/*</file>
                <mode>644</mode>
            </permission>
        </permissions>
    </extraDirectories>
</configuration>

Common Use Cases:

  • Add configuration files (application.yml, logback.xml)
  • Include startup scripts or health check scripts
  • Add SSL certificates or keystores
  • Include static resources or templates
  • Add monitoring agents (APM tools, Prometheus exporters)

APM Agent Integration (e.g., New Relic, Datadog)

<extraDirectories>
    <paths>
        <path>
            <from>agents</from>
            <into>/app/agents</into>
        </path>
    </paths>
</extraDirectories>
<container>
    <jvmFlags>
        <jvmFlag>-javaagent:/app/agents/newrelic-agent.jar</jvmFlag>
    </jvmFlags>
    <environment>
        <NEW_RELIC_APP_NAME>myapp</NEW_RELIC_APP_NAME>
        <NEW_RELIC_LICENSE_KEY>${env.NEW_RELIC_KEY}</NEW_RELIC_LICENSE_KEY>
    </environment>
</container>

Custom Startup Script

<extraDirectories>
    <paths>
        <path>
            <from>scripts</from>
            <into>/app/scripts</into>
        </path>
    </paths>
    <permissions>
        <permission>
            <file>/app/scripts/startup.sh</file>
            <mode>755</mode>
        </permission>
    </permissions>
</extraDirectories>
<container>
    <entrypoint>
        <shell>sh</shell>
        <option>-c</option>
        <arg>/app/scripts/startup.sh</arg>
    </entrypoint>
</container>

Native Library Support

<extraDirectories>
    <paths>
        <path>
            <from>src/main/native-libs</from>
            <into>/app/native</into>
        </path>
    </paths>
</extraDirectories>
<container>
    <jvmFlags>
        <jvmFlag>-Djava.library.path=/app/native</jvmFlag>
    </jvmFlags>
</container>
