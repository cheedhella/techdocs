8. Complete Configuration Example

Production-Ready Jib Configuration

A production-ready Jib configuration with all common options:

<plugin>
    <groupId>com.google.cloud.tools</groupId>
    <artifactId>jib-maven-plugin</artifactId>
    <version>3.4.0</version>
    <configuration>
        <!-- Base Image -->
        <from>
            <image>gcr.io/distroless/java17-debian12</image>
        </from>
        
        <!-- Target Image -->
        <to>
            <image>ghcr.io/myorg/myapp</image>
            <tags>
                <tag>latest</tag>
                <tag>${project.version}</tag>
                <tag>${git.commit.id.abbrev}</tag>
            </tags>
            <auth>
                <username>${env.GITHUB_ACTOR}</username>
                <password>${env.GITHUB_TOKEN}</password>
            </auth>
        </to>
        
        <!-- Container Configuration -->
        <container>
            <jvmFlags>
                <jvmFlag>-Xms256m</jvmFlag>
                <jvmFlag>-Xmx512m</jvmFlag>
                <jvmFlag>-XX:+UseContainerSupport</jvmFlag>
                <jvmFlag>-XX:MaxRAMPercentage=75.0</jvmFlag>
                <jvmFlag>-XX:+UseG1GC</jvmFlag>
                <jvmFlag>-Djava.security.egd=file:/dev/./urandom</jvmFlag>
            </jvmFlags>
            
            <mainClass>com.example.Application</mainClass>
            
            <args>
                <arg>--spring.profiles.active=prod</arg>
            </args>
            
            <ports>
                <port>8080</port>
                <port>9090</port>
            </ports>
            
            <environment>
                <SPRING_PROFILES_ACTIVE>production</SPRING_PROFILES_ACTIVE>
                <TZ>UTC</TZ>
            </environment>
            
            <labels>
                <maintainer>devops@example.com</maintainer>
                <version>${project.version}</version>
                <org.opencontainers.image.title>${project.artifactId}</org.opencontainers.image.title>
                <org.opencontainers.image.version>${project.version}</org.opencontainers.image.version>
                <org.opencontainers.image.created>${maven.build.timestamp}</org.opencontainers.image.created>
                <org.opencontainers.image.source>https://github.com/myorg/myapp</org.opencontainers.image.source>
            </labels>
            
            <user>nonroot:nonroot</user>
            <workingDirectory>/app</workingDirectory>
            <format>OCI</format>
            <creationTime>USE_CURRENT_TIMESTAMP</creationTime>
            
            <volumes>
                <volume>/tmp</volume>
            </volumes>
        </container>
        
        <!-- Extra Files -->
        <extraDirectories>
            <paths>
                <path>
                    <from>src/main/jib</from>
                </path>
            </paths>
            <permissions>
                <permission>
                    <file>/app/scripts/*.sh</file>
                    <mode>755</mode>
                </permission>
            </permissions>
        </extraDirectories>
        
        <!-- Output Paths -->
        <outputPaths>
            <digest>${project.build.directory}/jib-image.digest</digest>
            <imageId>${project.build.directory}/jib-image.id</imageId>
        </outputPaths>
        
        <!-- Allow insecure registries (dev only) -->
        <allowInsecureRegistries>false</allowInsecureRegistries>
    </configuration>
</plugin>

Profile-Based Configuration

<profiles>
    <!-- Development Profile -->
    <profile>
        <id>dev</id>
        <build>
            <plugins>
                <plugin>
                    <groupId>com.google.cloud.tools</groupId>
                    <artifactId>jib-maven-plugin</artifactId>
                    <configuration>
                        <to>
                            <image>localhost:5000/myapp</image>
                        </to>
                        <container>
                            <jvmFlags>
                                <jvmFlag>-Xdebug</jvmFlag>
                                <jvmFlag>-Xrunjdwp:transport=dt_socket,server=y,suspend=n,address=*:5005</jvmFlag>
                            </jvmFlags>
                            <ports>
                                <port>8080</port>
                                <port>5005</port>
                            </ports>
                        </container>
                    </configuration>
                </plugin>
            </plugins>
        </build>
    </profile>
    
    <!-- Production Profile -->
    <profile>
        <id>prod</id>
        <build>
            <plugins>
                <plugin>
                    <groupId>com.google.cloud.tools</groupId>
                    <artifactId>jib-maven-plugin</artifactId>
                    <configuration>
                        <from>
                            <image>gcr.io/distroless/java17-debian12</image>
                        </from>
                        <to>
                            <image>prod-registry.io/myapp</image>
                            <tags>
                                <tag>${project.version}</tag>
                            </tags>
                        </to>
                        <container>
                            <jvmFlags>
                                <jvmFlag>-Xms512m</jvmFlag>
                                <jvmFlag>-Xmx1024m</jvmFlag>
                                <jvmFlag>-XX:+UseContainerSupport</jvmFlag>
                            </jvmFlags>
                            <user>nonroot:nonroot</user>
                            <creationTime>USE_CURRENT_TIMESTAMP</creationTime>
                        </container>
                    </configuration>
                </plugin>
            </plugins>
        </build>
    </profile>
</profiles>
