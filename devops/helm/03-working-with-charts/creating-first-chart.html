Creating Your First Chart (helm create)

1. Creating a New Chart
# Create a new chart with default structure
helm create mychart

# This creates:
mychart/
├── Chart.yaml
├── values.yaml
├── charts/
└── templates/
    ├── NOTES.txt
    ├── _helpers.tpl
    ├── deployment.yaml
    ├── service.yaml
    ├── ingress.yaml
    ├── serviceaccount.yaml
    ├── hpa.yaml
    └── tests/
        └── test-connection.yaml
2. Understanding the Generated Files

Chart.yaml: Chart metadata

apiVersion: v2
name: mychart
description: A Helm chart for Kubernetes
type: application
version: 0.1.0
appVersion: "1.16.0"

values.yaml: Default configuration (simplified)

replicaCount: 1
image:
  repository: nginx
  tag: ""
  pullPolicy: IfNotPresent
service:
  type: ClusterIP
  port: 80
3. Testing the Generated Chart
# Lint the chart
helm lint mychart

# Dry run installation
helm install test-release ./mychart --dry-run --debug

# Install the chart
helm install test-release ./mychart

# Check the release
helm list
kubectl get all

# Uninstall
helm uninstall test-release
4. Customizing Your Chart

Step 1: Update Chart.yaml

apiVersion: v2
name: myapp
description: My awesome application
type: application
version: 1.0.0
appVersion: "1.0.0"

  - web
  - backend

  - name: Your Name
    email: you@example.com

  tag: "1.0.0"

  type: LoadBalancer

5. Simplifying the Chart

Remove unnecessary files for a minimal chart:

# Keep only what you need
mychart/
├── Chart.yaml
├── values.yaml
└── templates/
    ├── deployment.yaml
    └── service.yaml

# Delete unused files
rm mychart/templates/ingress.yaml
rm mychart/templates/serviceaccount.yaml
rm mychart/templates/hpa.yaml
rm -rf mychart/templates/tests/
6. Creating a Minimal Chart from Scratch
# Create directory structure
mkdir -p myapp/{templates,charts}
cd myapp

# Create Chart.yaml
cat <<EOF > Chart.yaml
apiVersion: v2
name: myapp
version: 1.0.0
appVersion: "1.0.0"
description: My application
EOF

# Create values.yaml
cat <<EOF > values.yaml
replicaCount: 2
image:
  repository: nginx
  tag: "1.21"
service:
  port: 80
EOF

# Create deployment template
cat <<EOF > templates/deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: {{ .Release.Name }}
spec:
  replicas: {{ .Values.replicaCount }}
  selector:
    matchLabels:
      app: {{ .Release.Name }}
  template:
    metadata:
      labels:
        app: {{ .Release.Name }}
    spec:
      containers:
      - name: {{ .Chart.Name }}
        image: "{{ .Values.image.repository }}:{{ .Values.image.tag }}"
        ports:
        - containerPort: {{ .Values.service.port }}
EOF

# Create service template
cat <<EOF > templates/service.yaml
apiVersion: v1
kind: Service
metadata:
  name: {{ .Release.Name }}
spec:
  type: ClusterIP
  ports:
  - port: {{ .Values.service.port }}
    targetPort: {{ .Values.service.port }}
  selector:
    app: {{ .Release.Name }}
EOF
7. Adding NOTES.txt
cat <<EOF > templates/NOTES.txt
Thank you for installing {{ .Chart.Name }}!

Your release is named {{ .Release.Name }}.

To access your application:

  export POD_NAME=$(kubectl get pods -n {{ .Release.Namespace }} -l "app={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  kubectl port-forward $POD_NAME 8080:{{ .Values.service.port }}
  echo "Visit http://127.0.0.1:8080"
EOF
8. Validating Your Chart
# Lint the chart
helm lint ./myapp

# Template rendering (see generated YAML)
helm template myrelease ./myapp

# Dry run with debug
helm install myrelease ./myapp --dry-run --debug

# Check for syntax errors
helm template myrelease ./myapp | kubectl apply --dry-run=client -f -
9. Installing Your Chart
# Install the chart
helm install myrelease ./myapp

# Verify installation
helm status myrelease
kubectl get all -l app=myrelease

# Test the application
kubectl port-forward svc/myrelease 8080:80
curl http://localhost:8080
10. Iterating on Your Chart
# Make changes to templates or values
vim myapp/values.yaml

# Test changes
helm upgrade myrelease ./myapp --dry-run --debug

# Apply changes
helm upgrade myrelease ./myapp

# Rollback if needed
helm rollback myrelease
11. Best Practices for New Charts
  • Start simple: Begin with minimal templates, add complexity as needed;
  • Use helm create: Good starting point, remove what you don't need;
  • Test frequently: Use helm lint and --dry-run often;
  • Document everything: Add comments in templates and comprehensive README;
  • Use meaningful names: Clear, descriptive names for charts and resources;
  • Follow conventions: Use standard Kubernetes labels and annotations;
