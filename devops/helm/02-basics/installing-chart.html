Installing a Chart

1. Basic Installation
# Install from local directory
helm install myrelease ./mychart

# Install from packaged chart
helm install myrelease mychart-0.1.0.tgz

# Install from repository
helm install myrelease bitnami/nginx

# Install with auto-generated name
helm install --generate-name bitnami/nginx
2. Installation Options
Option Description Example
--namespace Install in specific namespace helm install myapp ./chart --namespace production
--create-namespace Create namespace if it doesn't exist helm install myapp ./chart --create-namespace -n prod
--values / -f Override values from file helm install myapp ./chart -f values-prod.yaml
--set Override individual values helm install myapp ./chart --set image.tag=v2.0
--dry-run Simulate installation helm install myapp ./chart --dry-run
--wait Wait for resources to be ready helm install myapp ./chart --wait --timeout 5m
--atomic Rollback on failure helm install myapp ./chart --atomic
3. Overriding Values

Using --set:

# Single value
helm install myapp ./chart --set replicaCount=3

# Multiple values
helm install myapp ./chart --set replicaCount=3 --set image.tag=v2.0

# Nested values
helm install myapp ./chart --set image.repository=nginx --set image.tag=1.21

# Array values
helm install myapp ./chart --set ingress.hosts[0].host=example.com

Using values file:

# Single values file
helm install myapp ./chart -f values-prod.yaml

# Multiple values files (later files override earlier)
helm install myapp ./chart -f values-common.yaml -f values-prod.yaml

# Combine --set and -f (--set takes precedence)
helm install myapp ./chart -f values-prod.yaml --set image.tag=v3.0
4. Installation from Different Sources

From local directory:

helm install myapp ./mychart

From packaged chart:

helm install myapp mychart-1.0.0.tgz

From repository:

helm install myapp bitnami/nginx

From URL:

helm install myapp https://example.com/charts/mychart-1.0.0.tgz

From OCI registry:

helm install myapp oci://registry.example.com/charts/mychart --version 1.0.0
5. Dry Run - Test Before Installing
# See what would be installed without actually installing
helm install myapp ./chart --dry-run --debug

# Output shows:
# - Rendered templates
# - Values used
# - Any errors in templates
6. Waiting for Resources
# Wait for all pods to be ready (default timeout: 5m)
helm install myapp ./chart --wait

# Custom timeout
helm install myapp ./chart --wait --timeout 10m

# Atomic install (rollback if fails)
helm install myapp ./chart --atomic --timeout 5m
7. Installing Specific Version
# Install specific chart version
helm install myapp bitnami/nginx --version 13.2.0

# List available versions
helm search repo bitnami/nginx --versions
8. Post-Installation
# View installation status
helm status myapp

# List all releases
helm list

# View deployed resources
kubectl get all -l app.kubernetes.io/instance=myapp

# View release values
helm get values myapp

# View release manifest
helm get manifest myapp
9. Common Installation Patterns

Development environment:

helm install myapp ./chart \
  --namespace dev \
  --create-namespace \
  -f values-dev.yaml \
  --set image.tag=latest

Production environment:

helm install myapp ./chart \
  --namespace production \
  --create-namespace \
  -f values-prod.yaml \
  --set image.tag=v1.0.0 \
  --wait \
  --timeout 10m \
  --atomic
10. Troubleshooting Installation

Issue: Release already exists

# Error: cannot re-use a name that is still in use
# Solution: Use a different name or uninstall first
helm uninstall myapp
helm install myapp ./chart

Issue: Template rendering errors

# Use --debug to see detailed error messages
helm install myapp ./chart --dry-run --debug

Issue: Resources not ready

# Check pod status
kubectl get pods -n <namespace>

# Check pod logs
kubectl logs <pod-name> -n <namespace>

# Describe pod for events
kubectl describe pod <pod-name> -n <namespace>
