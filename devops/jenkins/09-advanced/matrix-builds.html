Matrix Builds

1. What?
  • Run same stages across multiple combinations of parameters;
  • Test on different JDK versions, OS, browsers;
2. Example
pipeline {
    agent none
    stages {
        stage("Test Matrix") {
            matrix {
                axes {
                    axis { name "JDK"; values "11","17","21" }
                    axis { name "OS"; values "linux","windows" }
                }
                stages {
                    stage("Test") {
                        agent { label "" }
                        steps { sh "java -version && mvn test" }
                    }
                }
            }
        }
    }
}
