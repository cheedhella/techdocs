Deploy App to Kubernetes Cluster

Jenkinsfile
pipeline {
    agent any
    environment { IMAGE = "registry.example.com/myapp" }
    stages {
        stage("Build and Push") {
            steps {
                sh "docker build -t \:\ ."
                sh "docker push \:"
            }
        }
        stage("Deploy") {
            steps {
                withCredentials([file(credentialsId: "kubeconfig", variable: "KUBECONFIG")]) {
                    sh "kubectl set image deployment/myapp myapp=\:"
                    sh "kubectl rollout status deployment/myapp --timeout=120s"
                }
            }
        }
        stage("Verify") {
            steps {
                withCredentials([file(credentialsId: "kubeconfig", variable: "KUBECONFIG")]) {
                    sh "kubectl get pods -l app=myapp"
                }
            }
        }
    }
}
