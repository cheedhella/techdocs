Implement Blue-Green Deployment

1. Concept
  • Two identical environments: Blue (current) and Green (new);
  • Deploy to Green, test, then switch traffic;
  • Instant rollback by switching back to Blue;
2. Jenkinsfile
pipeline {
    agent any
    environment { IMAGE = "registry.example.com/myapp" }
    stages {
        stage("Build") {
            steps {
                sh "docker build -t \:\ ."
                sh "docker push \:"
            }
        }
        stage("Deploy Green") {
            steps {
                sh "kubectl set image deployment/myapp-green myapp=\:"
                sh "kubectl rollout status deployment/myapp-green"
            }
        }
        stage("Smoke Test") {
            steps { sh "curl -f http://myapp-green.internal/health" }
        }
        stage("Switch Traffic") {
            input { message "Switch traffic to Green?" }
            steps {
                sh "kubectl patch service myapp -p '{ "spec": { "selector": { "version": "green" } } }'" 
            }
        }
    }
}
