Build and Deploy a Java App Using Maven

Jenkinsfile
pipeline {
    agent { docker { image "maven:3.9-eclipse-temurin-17" } }
    stages {
        stage("Checkout") {
            steps { git "https://github.com/user/java-app.git" }
        }
        stage("Build") {
            steps { sh "mvn clean package -DskipTests" }
        }
        stage("Test") {
            steps { sh "mvn test" }
            post { always { junit "**/target/surefire-reports/*.xml" } }
        }
        stage("Archive") {
            steps { archiveArtifacts artifacts: "target/*.jar" }
        }
    }
}
