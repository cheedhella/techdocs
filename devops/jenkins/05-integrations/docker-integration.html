Docker Integration

1. Docker in Jenkins Pipelines
  • Build Docker images, push to registries, use Docker as build agents;
  • Requires Docker Pipeline plugin and Docker installed on agents;
2. Build and Push Docker Image
pipeline {
    agent any
    environment {
        DOCKER_CREDS = credentials('docker-hub-creds')
    }
    stages {
        stage('Build Image') {
            steps {
                sh 'docker build -t myapp:${BUILD_NUMBER} .'
            }
        }
        stage('Push Image') {
            steps {
                sh 'echo $DOCKER_CREDS_PSW | docker login -u $DOCKER_CREDS_USR --password-stdin'
                sh 'docker push user/myapp:${BUILD_NUMBER}'
            }
        }
    }
}
