CI/CD to Kubernetes Cluster

1. Full Pipeline
pipeline {
    agent {
        kubernetes {
            yaml """
apiVersion: v1
kind: Pod
spec:
  containers:
  - name: maven
    image: maven:3.9
    command: ["sleep","99999"]
  - name: kubectl
    image: bitnami/kubectl
    command: ["sleep","99999"]
"""
        }
    }
    stages {
        stage("Build") {
            steps { container("maven") { sh "mvn clean package" } }
        }
        stage("Deploy") {
            steps {
                container("kubectl") {
                    sh "kubectl set image deployment/myapp myapp=reg/myapp:"
                    sh "kubectl rollout status deployment/myapp"
                }
            }
        }
    }
}
