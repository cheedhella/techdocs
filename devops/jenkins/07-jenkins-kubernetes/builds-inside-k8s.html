Running Builds Inside K8s

1. Multi-Container
pipeline {
    agent {
        kubernetes {
            yaml """
apiVersion: v1
kind: Pod
spec:
  containers:
  - name: node
    image: node:20
    command: ["sleep","99999"]
  - name: docker
    image: docker:24
    command: ["sleep","99999"]
"""
        }
    }
    stages {
        stage("Test") {
            steps { container("node") { sh "npm test" } }
        }
        stage("Build Image") {
            steps { container("docker") { sh "docker build -t myapp ." } }
        }
    }
}
