# Container Definition

name: c1                            # Name of the container, must be unique with in a Pod. You can't update it, once container is created.
image: nginx:1.10                   # Name and tag of Docker Image.
imagePullPolicy: Always                 # Image Pull Policy: It can be Always, Never, IfNotPresent.
                                        # Defaults to Always if tag is :latest. Otherwise, IfNotPresent.
                                        # Always - always pull the image from docker registry
                                        # IfNotPresent - pull only if that image is not present in local cache
                                        # Otherwise -  
env:                                # List of environment variables to set in the Container.
  - name: JDK_VERSION
    value: "1.8"
  - name: Ext_VERSION
    value: "6.0.2"

command:                            # It overrides the default command provided by docker image.
  - "/usr/bin/myprog"               # Command provided is not executed within a shell. 
  - "--host-port=asdf:2323"         # If you want to execute a command in a shell, do this: 
                                    #       command: ["/bin/sh"]
                                    #       args: ["-c", "while true; do echo hello; sleep 10;done"]

args:                           # It defines arguments to the above command. 
                                # If you define args without a command, default command is used with new args.

lifecycle:
  postStart:                                                      # Called immediately after a container is created. If the handler fails, container
    exec:                                                         # is terminated and restarted according to restart policy of ???.
      command: ["/bin/sh", "-c", "Hello from postStart handler"]
  preStop:                                                        # Called immediately before a container is terminated. Reason for the termination 
    exec:                                                         # is passed to the Handler. Container is terminated, regardless of the outcome 
      command: ["/usr/sbin/nginx","-s","quit"]                    # of the handler.

ports:                                   # List of ports to expose from the container.
  - containerPort: 81                    # It specifies which port to expose on the pod's IP address. 0 < x < 65536 
    hostPort: 82                         # It specifies which port to expose on the host.
    protocol: TCP                        # It can be TCP, UDP. Defaults to TCP.

volumeMounts:                 # It mounts specified volume at the specified location inside the Container.
  - mountPath: /var/data      # Container location where the volume is mounted. Anything written to this volume is saved at this path in the container.
    name: jenkins-home        # Name of the volume(as per Pod.volumes section)
    readOnly: true

            cpu: 500m

            cpu: 500m
            memory: 1500Mi

      - name: jenkins-home
        gcePersistentDisk:
          pdName: jenkins-home

apiVersion: v1
kind: Pod
metadata:
  name: my-nginx-pod
spec:
# It claims emptyDir volume and names it my-nginx-pod-vol1
  volumes:
  - name: my-nginx-pod-vol1
    emptyDir: {}
  containers:
  - name: my-nginx-pod-c1
    image: nginx
# volumeMounts: it specifies the mount path(anything written to specified volume is saved at this path in the container).
    volumeMounts:
    - name: my-nginx-pod-vol1
      mountPath: /usr/share/nginx/html
