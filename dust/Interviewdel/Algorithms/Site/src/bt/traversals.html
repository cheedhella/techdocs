Binary Tree Traversals
Recurisve Pre-Order 
                            
// Pre-order: root, left subtree, right subtree;
public void traversePreOrder(Node node) {
  if(node == null) return;
  System.out.print(" " + node.getValue());
  traversePreOrder(node.getLeftChild());
  traversePreOrder(node.getRightChild());
}
Non-Recurisve Pre-Order 
                        
public void nonRecursivePreOrder(Node root) {
	Stack s = new Stack();
	s.push(root);
	while (!s.isEmpty()) {
	    Node current = s.pop();
		System.out.print(" " + current.value);
		if (current.right != null) {
			s.push(current.right);
		}
		if (current.left != null) {
			s.push(current.left);
		}
	}
}
Recurisve In-Order 
                            
// In-order: left subtree, root, right subtree;
public void traverseInOrder(Node node) {
  if(node == null) return;
  traverseInOrder(node.getLeftChild());   
  System.out.print(" " + node.getValue());
  traverseInOrder(node.getRightChild()); 
}
Non-Recurisve In-Order 
                        
public void nonRecursiveInOrder(Node root) {
    Stack s = new Stack();
    Node current = root;
    while (current != null || !s.isEmpty()) {
        // Goto left most node!
        if (current != null) {
            s.push(current);
            current = current.left;
        } else {
            current = s.pop();
            // Print current node!
            System.out.print(" " + current.value);
            // Move to right subtree!
            current = current.right;
        }
    }
}
Recurisve Post-Order 
                            
// Post-order: left subtree, right subtree, root;
public void traversePostOrder(Node node) {
  if(node == null) return;
  traversePostOrder(node.getLeftChild());    
  traversePostOrder(node.getRightChild());
  System.out.print(" " + node.getValue());
}
Non-Recurisve Post-Order 
                        
/*
 * The idea is to move down to the left most node using left pointer; 
 * While moving down, push root and root's right child to stack; 
 * Once we reach leftmost node,
 *     Print it, if it doesn’t have a right child; 
 *     If it has a right child, then change root so that the right child is processed before;
 */
public void nonRecursivePostOrder(Node root) {
    Stack s = new Stack();
    Node current = root;
    while (current != null || !s.isEmpty()) {
        // Goto left most node!
        if (current != null) {
            if (current.right != null) {
                s.push(current.right);
                s.push(current);
                current = current.left;
            } else {
                current = s.pop();
                // If popped item has a right child and it is same as element at stack top, process that before!
                if (current.right == s.peek()) {
                    Node temp = s.pop();
                    s.push(current);
                    current = temp;
                } else {
                    System.out.print(" " + current.value); // Print current node!
                    current = null;
                }
            }
        }
    }
}
Non-Recurisve Post-Order - using 2 stacks; 
                        
/*
    If you observe, post-order is just the reversal of pre-order traversal; 
    So, the idea is, use pre-order traversal and instead of printing it, push it to stack s2; 
    At the end, keep pop from s2 and print it;
*/
public void nonRecursivePostOrder2(Node root) {
    Stack s1 = new Stack();
    Stack s2 = new Stack();
    Node current = root;
    while (current != null || !s1.isEmpty()) {
        // Goto left most node!
        if (current != null) {
            s1.push(current);
            current = current.left;
        } else {
            current = s1.pop();
            s2.push(current);
            // Move to right subtree!
            current = current.right; 
        }
    }
    while (!s2.isEmpty()) {
        System.out.print(" " + s2.pop().value);
    }
}
Breadth First Traversal 
                        
/*
 * BFS - It starts at the root and explores the neighbor 
 * nodes first, before moving to the next level neighbors; 
 */
public static void bfs(Node root) {
    Queue q = new LinkedList();
    q.add(root);
    while (!q.isEmpty()) {
        Node current = q.remove();
        if (current.left != null) 
            q.add(current.left);
        if (current.right != null) 
            q.add(current.right);
        System.out.print(" " + current.value);
    }
}
Depth First Traversal 
                        
/*
 * DFS - It starts at the root and explores as far as 
 * possible along each branch, before backtracking;
 */
public static void dfs(Node root) {
    Stack s = new Stack();
    s.push(root);
    while (!s.isEmpty()) {
        Node current = s.pop();
        if (current.right != null)
            s.push(current.right);
        if (current.left != null)
            s.push(current.left);
        System.out.print(" " + current.value);
    }
}
