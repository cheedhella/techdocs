• Given an array, find the sum of contiguous subarray with the largest sum?
        
import java.util.Arrays;

public class P7 {
    private static int[] rearrange(int[] data) {
        int[] output = new int[data.length];
        int i = 0;
        while (i < data.length / 2) {
            output[2 * i] = data[data.length - 1 - i];
            output[2 * i + 1] = data[i];
            i++;
        }

        if (data.length % 2 != 0) {
            output[2 * i] = data[i];
        }
        return output;

    }

    public static void main(String[] args) {
        int[] data = { 1, 2, 3, 4, 5 };
        int[] output = rearrange(data);
        System.out.println("After re-arranging: " + Arrays.toString(output));

        int[] data2 = { 1, 2, 3, 4, 5, 6 };
        int[] output2 = rearrange(data2);
        System.out.println("After re-arranging: " + Arrays.toString(output2));
    }
}

// Output:
After re-arranging: [5, 1, 4, 2, 3]
After re-arranging: [6, 1, 5, 2, 4, 3]
