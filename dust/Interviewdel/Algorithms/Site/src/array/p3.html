import java.util.OptionalInt;
import java.util.stream.IntStream;

public class P3 {
    public static void main(String[] args) {
        int[] a1 = { 9, 2, 3, 6 };
        OptionalInt optionalInt = IntStream.of(a1).min();
        if (optionalInt.isPresent())
            System.out.println("Min: " + optionalInt.getAsInt());
    }
}

// Output:
Min: 2
