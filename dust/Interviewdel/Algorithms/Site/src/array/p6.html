• Given an array, can you re-arrange its elements in such a way that the negative elements appear at one side and positive elements appear at the other?
        
import java.util.Arrays;

public class P6 {
    private static void rearrange(int[] data) {
        int lIndex = 0, rIndex = data.length - 1;
        while (lIndex < rIndex) {
            while (data[lIndex] < 0)
                lIndex++;
            while (data[rIndex] > 0)
                rIndex--;

            int temp = data[lIndex];
            data[lIndex] = data[rIndex];
            data[rIndex] = temp;
        }
    }

    public static void main(String[] args) {
        int[] data = { 10, -1, 20, 4, 5, -9, -6 };
        rearrange(data);
        System.out.println("After re-arranging: " + Arrays.toString(data));
    }
}

// Output:
After re-arranging: [-6, -1, 4, -9, 5, 20, 10]
