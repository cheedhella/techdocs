        Trie trie = getTrie();

        sortedList.forEach(s -> System.out.println(s));
    }

    private static Trie getTrie() {
        Trie d1 = new Trie();

        d1.insert("a");
        d1.insert("there");
        d1.insert("answer");
        d1.insert("any");
        d1.insert("by");
        d1.insert("bye");
        d1.insert("their");
        return d1;
    }
