Problem: Given an array of integers, find 2 pairs such their sum is equal? a + b = c + d 
            
import java.util.HashMap;
import java.util.Map;

public class FindPairApp {

    // Take a HashMap to store ;
    // Sum each number in the array with every other number and check hashmap to see if there already exists a pair with that sum;
    // Time Complexity: O(n^2);
    private static String findPair(int[] data) {
        Map sumMap = new HashMap<>();
        for (int i = 0; i < data.length; i++) {
            for (int j = i + 1; j < data.length; j++) {
                int sum = data[i] + data[j];
                if (sumMap.containsKey(sum)) {
                    int[] anotherPair = sumMap.get(sum);
                    return "(" + data[i] + "," + data[j] + ")" + "," + "(" + anotherPair[0] + ","
                            + anotherPair[1] + ")";
                }
                else {
                    sumMap.put(sum, new int[] { data[i], data[j] });
                }
            }
        }
        return "Couldn't find a pair!";
    }

    public static void main(String[] args) {
        int[] arr = { 3, 4, 7, 1, 12, 9 };
        System.out.println(findPair(arr));
    }
}

Output:
(7,9),(4,12)
